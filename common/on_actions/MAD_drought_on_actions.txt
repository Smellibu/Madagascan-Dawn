# common/on_actions/MAD_drought_on_actions.txt
# Madagascar startup and periodic mechanics.
# Tag-specific daily/monthly hooks prevent MAD effects from being run once for every country.

on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { country_exists = MAD }
				MAD = {
					# Older versions disabled MD protests immediately at startup. Keep them enabled
					# until the modern pressure idea is actually issued; preserve suppression in
					# migrated saves where the catalyst has already begun.
					if = {
						limit = {
							original_tag = MAD
							NOT = { has_country_flag = MAD_MODERN_PROTEST_CATALYST_ISSUED }
					}
						clr_country_flag = disabled_protests_system
						clr_country_flag = MAD_MODERN_MD_PROTESTS_INITIALISED
					}
					# Repair saves from the earlier permanent-suppression implementation.
					if = {
						limit = {
							original_tag = MAD
							OR = {
								has_country_flag = MAD_MODERN_PROTEST_CATALYST_FAILED
								has_country_flag = MAD_MODERN_CRISIS_COMPLETED
							}
						}
						clr_country_flag = disabled_protests_system
						clr_country_flag = MAD_TGV_CAPSAT_REVOLT_DEFERRED
					}
					# Migrate saves in which the modern crisis began before timer cancellation existed.
					if = {
						limit = {
							original_tag = MAD
							OR = {
								has_country_flag = MAD_MODERN_CRISIS_STARTED
								has_country_flag = MAD_MODERN_CRISIS_COMPLETED
							}
						}
						set_country_flag = MAD_TGV_CAPSAT_REVOLT_TIMER_CANCELLED
						clr_country_flag = MAD_TGV_CAPSAT_REVOLT_DEFERRED
					}
					MAD_dahalo_init = yes
					# Migrate old saves where the original one-step option represented both CNME and FIS.
					if = {
						limit = { has_country_flag = MAD_TGV_CNME_FIS_CREATED }
						set_country_flag = MAD_TGV_CNME_ESTABLISHED
						set_country_flag = MAD_TGV_FIS_ESTABLISHED
						set_country_flag = MAD_TGV_DST_ESTABLISHED
						remove_ideas = MAD_TGV_FIS_PARALLEL_SECURITY
					add_ideas = MAD_TGV_FIS_PARALLEL_SECURITY
					}
					MAD_rural_cooperation_initialize = yes
					# Mark the first eligible monthly tick independently of when the drought focus starts.
					if = {
						limit = {
							NOT = { has_idea = MAD_ARMY_INHERITED_BURDEN }
						NOT = { has_completed_focus = MAD_ARMY_LAND_FORCES_MODERNISATION_PLAN }
						}
						add_ideas = MAD_ARMY_INHERITED_BURDEN
					}
					if = {
						limit = { NOT = { has_country_flag = MAD_DROUGHT_STARTED } }
						MAD_drought_init = yes
						country_event = { id = mad_drought.2 days = 30 }
					}
					clamp_variable = { var = MAD_drought_prevention_count min = 0 max = 7 }
					# The hidden event synchronises party ideas first, then constitutional policy and the Rural Cooperation readout.
					country_event = mad_political.8
					# Arm the first-loss monitor once; it waits for the new government to form.
					if = {
						limit = {
							original_tag = MAD
							NOT = { has_country_flag = MAD_RATSIRAKA_FIRST_LOSS_DONE }
							NOT = { has_country_flag = MAD_RATSIRAKA_SUCCESSOR_SELECTED }
							NOT = { has_country_flag = MAD_AREMA_RENEWAL }
							NOT = { has_country_flag = MAD_AREMA_RECONCILIATION }
							NOT = { has_country_flag = MAD_RATSIRAKA_FIRST_LOSS_PENDING }
						}
						set_country_flag = MAD_RATSIRAKA_FIRST_LOSS_PENDING
						set_variable = { MAD_RATSIRAKA_FORMATION_STABLE_DAYS = 0 }
					}
				}
			}
		}
	}

	# This fires once per in-game day for MAD, not once per country worldwide.
	on_daily_MAD = {
		effect = {
			MAD_dahalo_daily_tick = yes
			MAD_drought_containment_daily_tick = yes
			MAD_autocomplete_forced_outcome_foci = yes
			MAD_JIRAMA_GENERATION_RECOVERY_DAILY = yes
			MAD_ratsiraka_check_government_formed = yes

			# 17 November 2006: General Fidy's challenge, only under a TIM government.
			if = {
				limit = {
					original_tag = MAD
					tag = MAD
					date > 2006.11.16
					date < 2006.11.18
					check_variable = { ruling_party = 2 }
					NOT = { has_country_flag = MAD_MIL_FIDY_2006_EVENT_FIRED }
				}
				set_country_flag = MAD_MIL_FIDY_2006_EVENT_FIRED
				set_country_flag = MAD_MIL_FIDY_2006_PENDING
				country_event = MAD_military_refoundation.7
			}

			# 3 November 2009 officer recruitment fallback.
			if = {
				limit = { original_tag = MAD tag = MAD date > 2009.11.2 date < 2009.11.4 NOT = { has_country_flag = MAD_HISTORICAL_2009_OFFICERS_RECRUITED } }
				set_country_flag = MAD_HISTORICAL_2009_OFFICERS_RECRUITED
			}

			# The Officers of March branch is visible only from 3 November through 31 December 2009.
			if = {
				limit = {
					original_tag = MAD
					date > 2009.11.2
					date < 2010.1.1
					NOT = { has_country_flag = MAD_MIL_FOCUS_TGV_MARCH_WINDOW_OPENED }
				}
				set_country_flag = MAD_MIL_FOCUS_TGV_MARCH_WINDOW_OPENED
				mark_focus_tree_layout_dirty = yes
			}
			if = {
				limit = {
					original_tag = MAD
					date > 2009.12.31
					NOT = { has_country_flag = MAD_MIL_FOCUS_TGV_MARCH_WINDOW_CLOSED }
				}
				set_country_flag = MAD_MIL_FOCUS_TGV_MARCH_WINDOW_CLOSED
				mark_focus_tree_layout_dirty = yes
			}

			# Richard Ravalomanana: 1 December 2012.
			if = {
				limit = { original_tag = MAD tag = MAD date > 2012.11.30 date < 2012.12.2 NOT = { has_country_flag = MAD_HISTORICAL_RICHARD_2012_PROMOTION } NOT = { has_country_flag = MAD_TGV_PROMOTIONS_2013_COMPLETED } }
				set_country_flag = MAD_HISTORICAL_RICHARD_2012_PROMOTION
				if = { limit = { has_character = MAD_richard_ravalomanana } MAD_richard_ravalomanana = { promote_leader = yes } }
			}

			# 1 January 2013: Tafara Edmond and Rafenomanjato Mamy Jerison.
			if = {
				limit = { original_tag = MAD tag = MAD date > 2012.12.31 date < 2013.1.2 NOT = { has_country_flag = MAD_HISTORICAL_2013_OFFICERS_RECRUITED } }
				set_country_flag = MAD_HISTORICAL_2013_OFFICERS_RECRUITED
				if = { limit = { NOT = { has_country_flag = MAD_TGV_PROMOTIONS_2013_COMPLETED } has_character = MAD_tafara_edmond } MAD_tafara_edmond = { promote_leader = yes } }
				if = { limit = { NOT = { has_country_flag = MAD_TGV_PROMOTIONS_2013_COMPLETED } has_character = MAD_rafenomanjato_mamy_jerison } MAD_rafenomanjato_mamy_jerison = { promote_leader = yes } }
			}

			# Béni Xavier Rasolofonirina: 1 September 2013.
			if = {
				limit = { original_tag = MAD tag = MAD date > 2013.8.31 date < 2013.9.2 NOT = { has_country_flag = MAD_HISTORICAL_BENI_2013_PROMOTION } NOT = { has_country_flag = MAD_TGV_PROMOTIONS_2013_COMPLETED } }
				set_country_flag = MAD_HISTORICAL_BENI_2013_PROMOTION
				if = { limit = { has_character = MAD_beni_xavier_rasolofonirina } MAD_beni_xavier_rasolofonirina = { promote_leader = yes } }
			}

			# The 25 February 2016 HVM recruitment list.
			if = {
				limit = { original_tag = MAD tag = MAD date > 2016.2.24 date < 2016.2.26 NOT = { has_country_flag = MAD_HISTORICAL_HVM_OFFICERS_RECRUITED } }
				set_country_flag = MAD_HISTORICAL_HVM_OFFICERS_RECRUITED
			}

			if = {
				limit = {
					original_tag = MAD
					date > 2021.7.19
					date < 2021.7.21
					MAD_TGV_IN_POWER = yes
					NOT = { has_country_flag = MAD_APOLLO21_EVENT_FIRED }
				}
				country_event = mad_apollo21.1
			}
			# When the player manually allows AREMA again, detect that party index 5 is no longer banned.
			if = {
				limit = {
					original_tag = MAD
					has_country_flag = MAD_AREMA_BANNED
					NOT = { has_country_flag = MAD_RATSIRAKA_RETURN_PENDING }
					NOT = { has_country_flag = MAD_RATSIRAKA_RETURN_RESOLVED }
					NOT = { check_variable = { party_banned_array^5 = 1 } }
				}
				clr_country_flag = MAD_AREMA_BANNED
				set_country_flag = MAD_RATSIRAKA_RETURN_PENDING
				country_event = mad_political.92
			}
			# RC readout is refreshed monthly and on ruling-party changes; avoid a daily recalculation.
		}
	}




	on_ruling_party_change = {
		effect = {
			if = {
				limit = { original_tag = MAD }
				# Ruling-party changes alter military and post-crisis focus visibility.
				mark_focus_tree_layout_dirty = yes
				# Reconcile party ideas first; the event applies the final constitutional model and refreshes Rural Cooperation.
				country_event = { id = mad_political.8 hours = 1 }
				set_variable = { MAD_RATSIRAKA_FORMATION_STABLE_DAYS = 0 }
			}
		}
	}
	on_civil_war_end = {
		effect = {
			if = {
				limit = { original_tag = MAD }
				# The modern military focus branch also depends on the civil war having ended.
				mark_focus_tree_layout_dirty = yes
			}
			if = {
				limit = { tag = MAD original_tag = MAD has_country_flag = MAD_MODERN_CIVIL_WAR_CAPITAL_MOVED owns_state = 267 }
				set_capital = { state = 267 remember_old_capital = no }
				clr_country_flag = MAD_MODERN_CIVIL_WAR_CAPITAL_MOVED
			}
		}
	}

}
