# INVESTOR CYCLE
#
# Three suitors (Comazar, Vecturis, Holding), each seated once for 150 days. The player picks who comes
# next (event mad_madarail.3). The seated suitor's own concession decisions (MAD_MADARAIL_CONCEDE_*) raise
# MAD_madarail_confidence and are remembered as offers (flags MAD_MADARAIL_OFFER_1/2/3). If the investors commit,
# MAD_madarail_enact_concessions builds the company's dynamic modifier from the operator's base terms + the offers.
# Confidence >= 75 lets the player commit (STAY) by completing the mission MAD_MADARAIL_INVESTOR_TENURE.
########################################################################

# Who is seated is tracked with flags (no ideas). The flag stays set after the investors commit, so the category
# text keeps showing the operator; it is cleared when a suitor leaves or the investors desist.
MAD_madarail_clear_operators = {
	clr_country_flag = MAD_MADARAIL_SEATED_COMAZAR
	clr_country_flag = MAD_MADARAIL_SEATED_VECTURIS
	clr_country_flag = MAD_MADARAIL_SEATED_HOLDING
}

MAD_madarail_clear_concessions = {
	clr_country_flag = MAD_MADARAIL_OFFER_1
	clr_country_flag = MAD_MADARAIL_OFFER_2
	clr_country_flag = MAD_MADARAIL_OFFER_3
}

# Concessions are only OFFERS until the investors commit. This is where they become real, all in one go:
# the seated company's dynamic modifier is created from the operator's base terms plus one slice per offered
# concession (so which investor it is, and what was conceded, decide the quality), and the one-off costs are paid
# (treasury / ownership / stability). Before this runs nothing lasting exists, and a suitor who leaves without
# committing leaves nothing behind. Slots 1/2/3 match the MAD_MADARAIL_CONCEDE_* decisions.
# Variables: MAD_<company>_<modifier>, read by the dynamic modifiers in common/dynamic_modifiers.
MAD_madarail_enact_concessions = {
	if = {
		limit = { has_country_flag = MAD_MADARAIL_SEATED_COMAZAR }

		if = {
			limit = { has_country_flag = MAD_MADARAIL_OFFER_1 }
			add_to_variable = { MAD_comazar_production_speed_rail_way_factor = 0.04 tooltip = production_speed_rail_way_factor_tt }
			add_to_variable = { MAD_comazar_production_speed_rail_terminal_factor = 0.03 tooltip = production_speed_rail_terminal_factor_tt }
			add_to_variable = { MAD_comazar_internal_investments_money_cost_modifier = 0.02 tooltip = internal_investments_money_cost_modifier_tt }
			set_temp_variable = { treasury_change = -5.00 }
			modify_treasury_effect = yes

		if = {
			limit = { has_country_flag = MAD_MADARAIL_OFFER_3 }

# Seats a suitor: the previous one (and its concessions) leaves, confidence resets to the newcomer's starting
# level, the 150-day mission starts. Comazar is the hardest to win over, the Holding consortium the easiest.
MAD_madarail_seat_comazar = {
	MAD_madarail_clear_operators = yes
	MAD_madarail_clear_concessions = yes
	set_country_flag = MAD_MADARAIL_COMAZAR_USED
	set_country_flag = MAD_MADARAIL_SEATED_COMAZAR
	set_variable = { MAD_madarail_confidence = 10 }
	add_to_variable = { MAD_madarail_stage = 1 }
	activate_mission = MAD_MADARAIL_INVESTOR_TENURE
}

MAD_madarail_seat_vecturis = {
	MAD_madarail_clear_operators = yes
	MAD_madarail_clear_concessions = yes
	set_country_flag = MAD_MADARAIL_VECTURIS_USED
	set_country_flag = MAD_MADARAIL_SEATED_VECTURIS
	set_variable = { MAD_madarail_confidence = 30 }
	add_to_variable = { MAD_madarail_stage = 1 }
	activate_mission = MAD_MADARAIL_INVESTOR_TENURE
}

MAD_madarail_seat_holding = {
	MAD_madarail_clear_operators = yes
	MAD_madarail_clear_concessions = yes
	set_country_flag = MAD_MADARAIL_HOLDING_USED
	set_country_flag = MAD_MADARAIL_SEATED_HOLDING
	set_variable = { MAD_madarail_confidence = 45 }
	add_to_variable = { MAD_madarail_stage = 1 }
	activate_mission = MAD_MADARAIL_INVESTOR_TENURE
}

# Outcome 1: the investors commit. The suitor in place stays and every concession OFFERED now takes effect; STAY focus unlocks.
MAD_madarail_apply_stay_outcome = {
	MAD_madarail_enact_concessions = yes
	set_country_flag = MAD_MADARAIL_OUTCOME_DECIDED
	set_country_flag = MAD_MADARAIL_STAY_UNLOCKED
	custom_effect_tooltip = MAD_madarail_stay_unlocked_tt
}

# Outcome 2: the investors leave (by the player's choice or because the third suitor ran out of time).
# Dire penalties start now; the Madarail state-ownership branch of the tree is what removes them.
MAD_madarail_apply_desist_outcome = {
	set_country_flag = MAD_MADARAIL_OUTCOME_DECIDED
	set_country_flag = MAD_MADARAIL_DESIST_UNLOCKED
	MAD_madarail_clear_operators = yes
	MAD_madarail_clear_concessions = yes
	if = {
		limit = { has_active_mission = MAD_MADARAIL_INVESTOR_TENURE }
		remove_mission = MAD_MADARAIL_INVESTOR_TENURE
	}
	add_ideas = MAD_MADARAIL_INVESTOR_EXODUS
	custom_effect_tooltip = MAD_madarail_desist_unlocked_tt
}

# Puts the right lease tier on the country for the current MAD_daewoo_viability:
#   < 30  TOXIC      30 - 59.99  TROUBLED      60+  SOUND
# Only swaps when the tier actually changes.
MAD_daewoo_update_tier = {
	if = {
		limit = {
			check_variable = { MAD_daewoo_viability < 30 }
			NOT = { has_idea = MAD_DAEWOO_LEASE_TOXIC }
		}
		MAD_daewoo_clear_tiers = yes
		add_ideas = MAD_DAEWOO_LEASE_TOXIC
	}
	if = {
		limit = {
			check_variable = { MAD_daewoo_viability > 29.99 }
			check_variable = { MAD_daewoo_viability < 60 }
			NOT = { has_idea = MAD_DAEWOO_LEASE_TROUBLED }
		}
		MAD_daewoo_clear_tiers = yes
		add_ideas = MAD_DAEWOO_LEASE_TROUBLED
	}
	if = {
		limit = {
			check_variable = { MAD_daewoo_viability > 59.99 }
			NOT = { has_idea = MAD_DAEWOO_LEASE_SOUND }
		}
		MAD_daewoo_clear_tiers = yes
		add_ideas = MAD_DAEWOO_LEASE_SOUND
	}
}

MAD_daewoo_clear_tiers = {
	if = { limit = { has_idea = MAD_DAEWOO_LEASE_TOXIC } remove_ideas = MAD_DAEWOO_LEASE_TOXIC }
	if = { limit = { has_idea = MAD_DAEWOO_LEASE_TROUBLED } remove_ideas = MAD_DAEWOO_LEASE_TROUBLED }
	if = { limit = { has_idea = MAD_DAEWOO_LEASE_SOUND } remove_ideas = MAD_DAEWOO_LEASE_SOUND }
}

# common/scripted_effects/MAD_daewoo_outcome_scripted_effects.txt
#
# Shared effects for the Daewoo outcomes: cancelled (MAD_DEAL_BROKEN), collapsed (event mad_daewoo.3 + focus MAD_DEAL_KEPT_FAILED)
# and signed off (MAD_DEAL_KEPT_MANAGED_WELL). Uses MAD_daewoo_clear_tiers from MAD_madarail_scripted_effects.txt.

# The concession is gone, whichever way it went: its cash crops stop feeding the drought and every agricultural
# complex needs more hands again.
MAD_daewoo_break_effects = {
	if = {
		limit = { has_country_flag = MAD_DAEWOO_KOREAN_AGRI_SECTOR }
		hidden_effect = {
			add_to_variable = { MAD_agri_productivity = -0.02 }
			add_to_variable = { MAD_agri_income_tax = -0.02 }
			force_update_dynamic_modifier = yes
			268 = {
				if = {
					limit = { agriculture_district > 0 }
					remove_building = {
						type = agriculture_district
						level = 1
					}
				}
			}
			clr_country_flag = MAD_DAEWOO_KOREAN_AGRI_SECTOR
		}
	}
	set_variable = { MAD_drought_daewoo_pressure = 0 }
	custom_effect_tooltip = {
		localization_key = modifies_dynamic_modifier_tt
		MODIFIER = MAD_agriculture_modifier
	}
	add_to_variable = { MAD_agri_worker_req = 0.03 tooltip = agriculture_district_worker_requirement_modifier_tt }
	force_update_dynamic_modifier = yes
}

# The lease ends: the tier idea goes (the Sound tier hands its farmland bonuses back in its on_remove) and the break effects run.
MAD_daewoo_end_lease = {
	MAD_daewoo_clear_tiers = yes
	MAD_daewoo_break_effects = yes
}


########################################################################
# DAEWOO ELECTION DISPATCHER
# The MD election campaign is deliberately queued from the daily tick rather than from a focus reward.
# This keeps MD4_election_campaign.0 out of the focus tooltip/effect evaluation path.
########################################################################

MAD_daewoo_election_daily_tick = {
	# Deal Broken: only call the election campaign when the government is below 50%.
	if = {
		limit = {
			has_country_flag = MAD_DAEWOO_ELECTION_PENDING_BROKEN
		}
		clr_country_flag = MAD_DAEWOO_ELECTION_PENDING_BROKEN
		if = {
			limit = { MAD_government_popularity_below_half = yes }
			set_politics = { elections_allowed = yes }
			hidden_effect = { country_event = { id = MD4_election_campaign.0 days = 0 } }
		}
	}

	# Deal Collapses: always start the full Millennium Dawn election campaign.
	if = {
		limit = {
			has_country_flag = MAD_DAEWOO_ELECTION_PENDING_COLLAPSE
		}
		clr_country_flag = MAD_DAEWOO_ELECTION_PENDING_COLLAPSE
		set_politics = { elections_allowed = yes }
		hidden_effect = { country_event = { id = MD4_election_campaign.0 days = 0 } }
	}
}

########################################################################
# FORCED OUTCOME FOCUSES
# These branches are consequences of the systems, not player choices.
# Complete them automatically as soon as their actual outcome conditions exist.
########################################################################

MAD_autocomplete_forced_outcome_foci = {
	# The drought has definitively failed: force the losing branch.
	if = {
		limit = {
			has_completed_focus = MAD_Fighting_climate_change
			has_country_flag = MAD_DROUGHT_FAILED_STATE
			NOT = { has_country_flag = MAD_DROUGHT_WON }
			NOT = { has_completed_focus = MAD_losing_the_fight }
			NOT = { has_completed_focus = MAD_Winning_the_fight }
		}
		complete_national_focus = MAD_losing_the_fight
	}

	# The drought is fully resolved after the losing branch: force Crop Death.
	if = {
		limit = {
			has_completed_focus = MAD_losing_the_fight
			MAD_drought_resolved = yes
			NOT = { has_completed_focus = MAD_crop_death }
		}
		complete_national_focus = MAD_crop_death
	}

	# The Daewoo lease has collapsed after being kept: force Deal Collapses.
	if = {
		limit = {
			has_completed_focus = MAD_DEAL_KEPT
			has_country_flag = MAD_DAEWOO_OUTCOME_FAILED
			NOT = { has_completed_focus = MAD_DEAL_KEPT_FAILED }
			NOT = { has_completed_focus = MAD_DEAL_KEPT_MANAGED_WELL }
		}
		complete_national_focus = MAD_DEAL_KEPT_FAILED
	}
}


# --- the drought hooks -------------------------------------------------------------------------------------------
# Korean cash crops feed the drought, but only while the drought exists. The monthly tick adds MAD_drought_daewoo_pressure
# to the severity every month (halved once the outgrower decision, flag MAD_DAEWOO_FOOD_QUOTA, is taken) and the whole
# term switches off the moment the drought is over.

MAD_daewoo_drought_kept = {
	if = {
		limit = { MAD_drought_exists = yes }
		custom_effect_tooltip = MAD_daewoo_drought_kept_tt
		add_to_variable = { MAD_drought_daewoo_pressure = 0.20 }
		set_temp_variable = { MAD_dr_change = 4 }
		MAD_drought_apply_change = yes
	}
}

# The success grows more cash crops on the extra estate, so it feeds the drought number harder.
MAD_daewoo_drought_well = {
	if = {
		limit = { MAD_drought_exists = yes }
		custom_effect_tooltip = MAD_daewoo_drought_well_tt
		add_to_variable = { MAD_drought_daewoo_pressure = 0.15 }
		set_temp_variable = { MAD_dr_change = 4 }
		MAD_drought_apply_change = yes
	}
}

# common/scripted_effects/MAD_drought_scripted_effects.txt
#
# THE DROUGHT AND FAMINE PANEL
#
# One stat: MAD_drought_severity (0-100). It exists from game start (event mad_drought.1) and is moved once a month by
# MAD_drought_monthly_tick (event mad_drought.2 re-fires itself every 30 days until the drought is over).
#
# Every month:   severity += base pressure (creeps up every year)
#                          + Grand Sud term  (0.30, cut by up to 45% by infrastructure in the Toliara state, never to 0)
#                          + West term       (0.15, same rule, scored on state 266)
#                          + Daewoo cash crops (only while the drought exists; halved by the outgrower decision)
#                          + lingering shocks (historical droughts, failed rains; decays 20% a month)
#                          - preventative measures (permanent; each preventative decision adds to the total)
#                          - a recovery bonus once 5+ measures are in place
#                          - 2.5 more once Winning The Fight is done
#   Emergency water supply cuts a POSITIVE monthly rise to 30% for 6 months, then a +4 rebound.
#
# Phases:
#   MAD_DROUGHT_CONTAINED_STATE  set automatically: 5+ preventative measures and severity <= 30   -> Winning The Fight opens
#   MAD_DROUGHT_FAILED_STATE     set automatically: exodus meter (months at severity 60+, net) reaches 12 -> Losing The Fight opens
#   MAD_DROUGHT_WON / _LOST      set by those two focuses
#   MAD_DROUGHT_OVER             severity reached 0 -> MAD_drought_resolved unlocks the further agricultural focuses

# Expects the temp variable MAD_dr_change (positive = worse, negative = better).
#
# COUNTRYWIDE change. Since the per-state map went in, the STATES hold the real numbers and the national
# MAD_drought_severity is their average - so a countrywide change is pushed into every state (scaled by that
# state's exposure: the Grand Sud feels everything harder) and then the national number is recomputed.
# Every existing national decision, event and focus that sets MAD_dr_change keeps working unchanged.
MAD_drought_apply_change = {
	# temp variables do not survive a scope change, so hand it over on a real country variable
	set_variable = { MAD_dr_change_v = MAD_dr_change }

	# The drought model covers exactly four Madagascar states. Do not apply the drought
	# system to unrelated owned territory: the panel average must equal these four states.
	266 = {
		set_temp_variable = { MAD_dr_st = ROOT.MAD_dr_change_v }
		multiply_temp_variable = { MAD_dr_st = MAD_state_drought_exposure }
		if = {
			limit = {
				check_variable = { MAD_state_relief_months > 0 }
				check_variable = { MAD_dr_st > 0 }
			}
			multiply_temp_variable = { MAD_dr_st = 0.3 }
		}
		add_to_variable = { MAD_state_drought = MAD_dr_st }
		MAD_drought_state_clamp = yes
	}
	267 = {
		set_temp_variable = { MAD_dr_st = ROOT.MAD_dr_change_v }
		multiply_temp_variable = { MAD_dr_st = MAD_state_drought_exposure }
		if = {
			limit = {
				check_variable = { MAD_state_relief_months > 0 }
				check_variable = { MAD_dr_st > 0 }
			}
			multiply_temp_variable = { MAD_dr_st = 0.3 }
		}
		add_to_variable = { MAD_state_drought = MAD_dr_st }
		MAD_drought_state_clamp = yes
	}
	268 = {
		set_temp_variable = { MAD_dr_st = ROOT.MAD_dr_change_v }
		multiply_temp_variable = { MAD_dr_st = MAD_state_drought_exposure }
		if = {
			limit = {
				check_variable = { MAD_state_relief_months > 0 }
				check_variable = { MAD_dr_st > 0 }
			}
			multiply_temp_variable = { MAD_dr_st = 0.3 }
		}
		add_to_variable = { MAD_state_drought = MAD_dr_st }
		MAD_drought_state_clamp = yes
	}
	487 = {
		set_temp_variable = { MAD_dr_st = ROOT.MAD_dr_change_v }
		multiply_temp_variable = { MAD_dr_st = MAD_state_drought_exposure }
		if = {
			limit = {
				check_variable = { MAD_state_relief_months > 0 }
				check_variable = { MAD_dr_st > 0 }
			}
			multiply_temp_variable = { MAD_dr_st = 0.3 }
		}
		add_to_variable = { MAD_state_drought = MAD_dr_st }
		MAD_drought_state_clamp = yes
	}
	MAD_drought_recalc_national = yes
	MAD_drought_states_update_tiers = yes
}
# LOCAL change to one state only. Scope into the state and set the temp variable MAD_dr_state_change first:
#   FROM = { set_temp_variable = { MAD_dr_state_change = -6 } MAD_drought_state_apply_change = yes }
# State scope.
MAD_drought_state_apply_change = {
	add_to_variable = { MAD_state_drought = MAD_dr_state_change }
	MAD_drought_state_clamp = yes
	MAD_drought_state_update_tier = yes
}

# LOCAL famine change, state scope. Set MAD_dr_state_famine first (negative = fed, positive = hungrier).
MAD_drought_state_apply_famine = {
	add_to_variable = { MAD_state_famine = MAD_dr_state_famine }
	clamp_variable = { var = MAD_state_famine min = 0 max = 100 }
	MAD_drought_state_update_tier = yes
}

MAD_drought_clamp = {
	clamp_variable = { var = MAD_drought_severity min = 0 max = 100 }
	# the drought cannot be talked away early: below 5 only once it is contained, lost or won
	if = {
		limit = {
			NOT = { has_country_flag = MAD_DROUGHT_CONTAINED_STATE }
			NOT = { has_country_flag = MAD_DROUGHT_FAILED_STATE }
			NOT = { has_country_flag = MAD_DROUGHT_WON }
		}
		clamp_variable = { var = MAD_drought_severity min = 5 max = 100 }
	}
}

# One preventative decision done: the temp variable MAD_dr_prevent_amount comes off the monthly pressure for good,
# and the number drops a little at once. Set MAD_dr_prevent_amount right before calling this (see note above on why
# HOI4 scripted effects use a temp variable here instead of a parameter).
MAD_drought_add_prevention = {
	# Only the seven formal drought decisions count toward the 5/7 containment gate. CNA measures have
	# their own continuing pressure-reduction pool so they cannot produce 10/7 again.
	if = {
		limit = { check_variable = { MAD_drought_prevention_count < 7 } }
		add_to_variable = { MAD_drought_prevention_total = MAD_dr_prevent_amount }
		add_to_variable = { MAD_drought_prevention_count = 1 }
		set_temp_variable = { MAD_dr_change = -3 }
		MAD_drought_apply_change = yes

	}
}

MAD_drought_clear_tiers = {
	if = { limit = { has_idea = MAD_DROUGHT_TIER_1 } remove_ideas = MAD_DROUGHT_TIER_1 }
	if = { limit = { has_idea = MAD_DROUGHT_TIER_2 } remove_ideas = MAD_DROUGHT_TIER_2 }
	if = { limit = { has_idea = MAD_DROUGHT_TIER_3 } remove_ideas = MAD_DROUGHT_TIER_3 }
	if = { limit = { has_idea = MAD_DROUGHT_TIER_4 } remove_ideas = MAD_DROUGHT_TIER_4 }
}

#   1-29  recurring dry spells     30-54 drought in the south     55-74 famine conditions     75+ kere (catastrophic famine)
MAD_drought_update_tier = {
	if = {
		limit = { check_variable = { MAD_drought_severity < 0.001 } }
		MAD_drought_clear_tiers = yes
	}
	else_if = {
		limit = {
			check_variable = { MAD_drought_severity < 30 }
			NOT = { has_idea = MAD_DROUGHT_TIER_1 }
		}
		MAD_drought_clear_tiers = yes
		add_ideas = MAD_DROUGHT_TIER_1
	}
	else_if = {
		limit = {
			check_variable = { MAD_drought_severity > 29.99 }
			check_variable = { MAD_drought_severity < 55 }
			NOT = { has_idea = MAD_DROUGHT_TIER_2 }
		}
		MAD_drought_clear_tiers = yes
		add_ideas = MAD_DROUGHT_TIER_2
	}
	else_if = {
		limit = {
			check_variable = { MAD_drought_severity > 54.99 }
			check_variable = { MAD_drought_severity < 75 }
			NOT = { has_idea = MAD_DROUGHT_TIER_3 }
		}
		MAD_drought_clear_tiers = yes
		add_ideas = MAD_DROUGHT_TIER_3
	}
	else_if = {
		limit = {
			check_variable = { MAD_drought_severity > 74.99 }
			NOT = { has_idea = MAD_DROUGHT_TIER_4 }
		}
		MAD_drought_clear_tiers = yes
		add_ideas = MAD_DROUGHT_TIER_4
	}
}

# INFRASTRUCTURE (why the Grand Sud suffers less with roads - but is never spared)

# Scores infrastructure (0-3) in the Toliara state (268, the Grand Sud) into MAD_drought_infra_south.
# HOI4 triggers can't take parameters either, so this is written out directly rather than as one generic
# "$VAR$/$STATE$" effect (that syntax is Stellaris/Vic3/CK3-only and is not valid HOI4 script).
# State-scope infrastructure trigger: infrastructure 3 = score 1, 4 = score 2, 5 = score 3.
MAD_drought_score_south = {
	set_variable = { MAD_drought_infra_south = 0 }
	if = { limit = { 268 = { infrastructure > 2 } } add_to_variable = { MAD_drought_infra_south = 1 } }
	if = { limit = { 268 = { infrastructure > 3 } } add_to_variable = { MAD_drought_infra_south = 1 } }
	if = { limit = { 268 = { infrastructure > 4 } } add_to_variable = { MAD_drought_infra_south = 1 } }
}

# Same scoring for Mahajanga (266) into MAD_drought_infra_west - the north-western term of the monthly pressure.
MAD_drought_score_west = {
	set_variable = { MAD_drought_infra_west = 0 }
	if = { limit = { 266 = { infrastructure > 2 } } add_to_variable = { MAD_drought_infra_west = 1 } }
	if = { limit = { 266 = { infrastructure > 3 } } add_to_variable = { MAD_drought_infra_west = 1 } }
	if = { limit = { 266 = { infrastructure > 4 } } add_to_variable = { MAD_drought_infra_west = 1 } }
}

MAD_drought_read_infrastructure = {
	MAD_drought_score_south = yes
	MAD_drought_score_west = yes
}

# Adds 0.30 x (1 - 0.15 x south infrastructure score) to the monthly pressure: always positive, whatever the roads are like.
MAD_drought_add_south_term = {
	set_temp_variable = { MAD_dr_term = MAD_drought_infra_south }
	multiply_temp_variable = { MAD_dr_term = -0.15 }
	add_to_temp_variable = { MAD_dr_term = 1 }
	multiply_temp_variable = { MAD_dr_term = 0.30 }
	add_to_temp_variable = { MAD_dr_delta = MAD_dr_term }
}

# Same for the west, weighted 0.15.
MAD_drought_add_west_term = {
	set_temp_variable = { MAD_dr_term = MAD_drought_infra_west }
	multiply_temp_variable = { MAD_dr_term = -0.15 }
	add_to_temp_variable = { MAD_dr_term = 1 }
	multiply_temp_variable = { MAD_dr_term = 0.15 }
	add_to_temp_variable = { MAD_dr_delta = MAD_dr_term }
}

# A drought that hits the Grand Sud: the temp variable MAD_dr_shock_base sets its severity, cut by infrastructure in
# the Toliara state (never to 0), and it lingers. Set MAD_dr_shock_base right before calling this.
MAD_drought_south_shock = {
	MAD_drought_read_infrastructure = yes
	set_temp_variable = { MAD_dr_change = MAD_drought_infra_south }
	multiply_temp_variable = { MAD_dr_change = -0.15 }
	add_to_temp_variable = { MAD_dr_change = 1 }
	multiply_temp_variable = { MAD_dr_change = MAD_dr_shock_base }
	MAD_drought_apply_change = yes
	add_to_variable = { MAD_drought_shock_pressure = 0.4 }
}

########################################################################
# INTERNAL DROUGHT MIGRATION
########################################################################
# Finds the most drought-stricken of the four modeled Madagascar states and moves a fixed
# number of people into the least-affected one. Historical shocks set MAD_drought_migration_force = 1,
# while normal monthly migration only happens when the drought gap is large enough.
#
# MAD_drought_migration_amount is in THOUSANDS for the panel/readout.
# add_manpower is used in state scope because it changes the state's total population.
# A negative state add_manpower also adds directly to recruitable manpower in HOI4, so the
# country-level correction below removes that side-effect from the national manpower pool.

MAD_drought_migrate_worst_to_best = {
	# Snapshot the four modeled state severities into country variables.
	266 = { set_variable = { ROOT.MAD_mig_d_266 = MAD_state_drought } }
	267 = { set_variable = { ROOT.MAD_mig_d_267 = MAD_state_drought } }
	268 = { set_variable = { ROOT.MAD_mig_d_268 = MAD_state_drought } }
	487 = { set_variable = { ROOT.MAD_mig_d_487 = MAD_state_drought } }

	set_variable = { MAD_mig_found = 0 }

	# Most affected = highest drought severity. Ties are resolved deterministically by state-id order.
	if = {
		limit = {
			NOT = { check_variable = { MAD_mig_d_266 < MAD_mig_d_267 } }
			NOT = { check_variable = { MAD_mig_d_266 < MAD_mig_d_268 } }
			NOT = { check_variable = { MAD_mig_d_266 < MAD_mig_d_487 } }
		}
		266 = { save_event_target_as = MAD_drought_migration_source }
		if = {
			limit = {
				check_variable = { MAD_mig_d_267 < MAD_mig_d_268 }
				check_variable = { MAD_mig_d_267 < MAD_mig_d_487 }
			}
			267 = { save_event_target_as = MAD_drought_migration_destination }
		}
		else_if = {
			limit = { check_variable = { MAD_mig_d_268 < MAD_mig_d_487 } }
			268 = { save_event_target_as = MAD_drought_migration_destination }
		}
		else = {
			487 = { save_event_target_as = MAD_drought_migration_destination }
		}
		set_variable = { MAD_mig_found = 1 }
	}
	else_if = {
		limit = {
			NOT = { check_variable = { MAD_mig_d_267 < MAD_mig_d_266 } }
			NOT = { check_variable = { MAD_mig_d_267 < MAD_mig_d_268 } }
			NOT = { check_variable = { MAD_mig_d_267 < MAD_mig_d_487 } }
		}
		267 = { save_event_target_as = MAD_drought_migration_source }
		if = {
			limit = {
				check_variable = { MAD_mig_d_266 < MAD_mig_d_268 }
				check_variable = { MAD_mig_d_266 < MAD_mig_d_487 }
			}
			266 = { save_event_target_as = MAD_drought_migration_destination }
		}
		else_if = {
			limit = { check_variable = { MAD_mig_d_268 < MAD_mig_d_487 } }
			268 = { save_event_target_as = MAD_drought_migration_destination }
		}
		else = {
			487 = { save_event_target_as = MAD_drought_migration_destination }
		}
		set_variable = { MAD_mig_found = 1 }
	}
	else_if = {
		limit = {
			NOT = { check_variable = { MAD_mig_d_268 < MAD_mig_d_266 } }
			NOT = { check_variable = { MAD_mig_d_268 < MAD_mig_d_267 } }
			NOT = { check_variable = { MAD_mig_d_268 < MAD_mig_d_487 } }
		}
		268 = { save_event_target_as = MAD_drought_migration_source }
		if = {
			limit = {
				check_variable = { MAD_mig_d_266 < MAD_mig_d_267 }
				check_variable = { MAD_mig_d_266 < MAD_mig_d_487 }
			}
			266 = { save_event_target_as = MAD_drought_migration_destination }
		}
		else_if = {
			limit = { check_variable = { MAD_mig_d_267 < MAD_mig_d_487 } }
			267 = { save_event_target_as = MAD_drought_migration_destination }
		}
		else = {
			487 = { save_event_target_as = MAD_drought_migration_destination }
		}
		set_variable = { MAD_mig_found = 1 }
	}
	else = {
		487 = { save_event_target_as = MAD_drought_migration_source }
		if = {
			limit = {
				check_variable = { MAD_mig_d_266 < MAD_mig_d_267 }
				check_variable = { MAD_mig_d_266 < MAD_mig_d_268 }
			}
			266 = { save_event_target_as = MAD_drought_migration_destination }
		}
		else_if = {
			limit = { check_variable = { MAD_mig_d_267 < MAD_mig_d_268 } }
			267 = { save_event_target_as = MAD_drought_migration_destination }
		}
		else = {
			268 = { save_event_target_as = MAD_drought_migration_destination }
		}
		set_variable = { MAD_mig_found = 1 }
	}

	if = {
		limit = {
			check_variable = { MAD_mig_found > 0.5 }
			check_variable = { MAD_drought_migration_amount > 0.001 }
		}
		# Read the selected states' current drought values before deciding whether monthly migration is large enough.
		event_target:MAD_drought_migration_source = {
			set_variable = { ROOT.MAD_mig_source_drought = MAD_state_drought }
		}
		event_target:MAD_drought_migration_destination = {
			set_variable = { ROOT.MAD_mig_destination_drought = MAD_state_drought }
		}
		set_temp_variable = { MAD_mig_gap = MAD_mig_source_drought }
		subtract_from_temp_variable = { MAD_mig_gap = MAD_mig_destination_drought }

		if = {
			limit = {
				OR = {
					check_variable = { MAD_drought_migration_force > 0.5 }
					AND = {
						check_variable = { MAD_drought_severe_states > 0.5 }
						check_variable = { MAD_mig_gap > 9.99 }
					}
				}
			}
			# Amount is stored in thousands for the UI; state population effects use people.
			set_temp_variable = { MAD_mig_people = MAD_drought_migration_amount }
			multiply_temp_variable = { MAD_mig_people = 1000 }
			set_temp_variable = { MAD_mig_negative_people = MAD_mig_people }
			multiply_temp_variable = { MAD_mig_negative_people = -1 }

			event_target:MAD_drought_migration_source = {
				add_manpower = var:MAD_mig_negative_people
			}
			event_target:MAD_drought_migration_destination = {
				add_manpower = var:MAD_mig_people
			}

			# Negative state add_manpower has a direct recruitable-manpower side effect; undo it nationally.
			add_manpower = var:MAD_mig_negative_people
			add_to_variable = { MAD_drought_displaced = MAD_drought_migration_amount }
		}
	}

	set_variable = { MAD_drought_migration_amount = 0 }
	set_variable = { MAD_drought_migration_force = 0 }
}

########################################################################
# THE MONTHLY TICK
########################################################################

MAD_drought_monthly_tick = {
	if = {
		limit = { MAD_drought_exists = yes }

		# the climate creeps: every twelve months the base pressure rises (until it is won)
		add_to_variable = { MAD_drought_months = 1 }
		add_to_variable = { MAD_drought_calendar_months = 1 }
		if = {
			limit = { check_variable = { MAD_drought_months > 11.99 } }
			set_variable = { MAD_drought_months = 0 }
			if = {
				limit = {
					check_variable = { MAD_drought_base_pressure < 0.65 }
					NOT = { has_country_flag = MAD_DROUGHT_WON }
				}
				add_to_variable = { MAD_drought_base_pressure = 0.03 }
			}
		}

		# history and chance
		MAD_drought_scheduled_events = yes

		# gross pressure this month
		set_temp_variable = { MAD_dr_delta = MAD_drought_base_pressure }
		MAD_drought_read_infrastructure = yes
		MAD_drought_add_south_term = yes
		MAD_drought_add_west_term = yes

		# Daewoo's cash crops - only while the drought exists (this whole block is inside that check)
		if = {
			limit = { check_variable = { MAD_drought_daewoo_pressure > 0 } }
			set_temp_variable = { MAD_dr_term = MAD_drought_daewoo_pressure }
			if = {
				limit = { has_country_flag = MAD_DAEWOO_FOOD_QUOTA }
				multiply_temp_variable = { MAD_dr_term = 0.5 }
			}
			add_to_temp_variable = { MAD_dr_delta = MAD_dr_term }
		}

		# lingering shocks
		add_to_temp_variable = { MAD_dr_delta = MAD_drought_shock_pressure }
		multiply_variable = { MAD_drought_shock_pressure = 0.8 }

		# what has been built to prevent it
		subtract_from_temp_variable = { MAD_dr_delta = MAD_drought_prevention_total }
		subtract_from_temp_variable = { MAD_dr_delta = MAD_drought_cna_pressure_reduction }
		# Migration guard for older saves where CNA work had been incorrectly counted as 10/7 formal measures.
		clamp_variable = { var = MAD_drought_prevention_count min = 0 max = 7 }
		if = {
			limit = { check_variable = { MAD_drought_prevention_count > 4.99 } }
			set_temp_variable = { MAD_dr_term = MAD_drought_prevention_count }
			subtract_from_temp_variable = { MAD_dr_term = 4 }
			multiply_temp_variable = { MAD_dr_term = 0.2 }
			subtract_from_temp_variable = { MAD_dr_delta = MAD_dr_term }
		}
		if = {
			limit = { has_country_flag = MAD_DROUGHT_WON }
			subtract_from_temp_variable = { MAD_dr_delta = 2.5 }
		}

		# apply it - through the states, which then set the national average
		set_variable = { MAD_drought_last_delta = MAD_dr_delta }
		set_temp_variable = { MAD_dr_change = MAD_dr_delta }
		MAD_drought_apply_change = yes
		# each state then drifts a little on its own and its famine meter follows its drought
		MAD_drought_state_monthly = yes

		# People leave the states that are actually in severe drought, not just because the national average is high.
		# The severe-state counter drives the exodus/failure logic; actual people are moved below.
		set_variable = { MAD_drought_severe_states = 0 }
		266 = { if = { limit = { check_variable = { MAD_state_drought > 59.99 } } ROOT = { add_to_variable = { MAD_drought_severe_states = 1 } } } }
		267 = { if = { limit = { check_variable = { MAD_state_drought > 59.99 } } ROOT = { add_to_variable = { MAD_drought_severe_states = 1 } } } }
		268 = { if = { limit = { check_variable = { MAD_state_drought > 59.99 } } ROOT = { add_to_variable = { MAD_drought_severe_states = 1 } } } }
		487 = { if = { limit = { check_variable = { MAD_state_drought > 59.99 } } ROOT = { add_to_variable = { MAD_drought_severe_states = 1 } } } }
		if = {
			limit = { check_variable = { MAD_drought_severe_states > 0.5 } }
			add_to_variable = { MAD_drought_depop = 1 }
			if = {
				limit = { NOT = { has_idea = MAD_DROUGHT_EVACUATIONS } }
				add_ideas = MAD_DROUGHT_EVACUATIONS
			}

			# 6,000 people per severe state-month, moved from the worst-hit modeled state to the least-affected one.
			set_variable = { MAD_drought_migration_amount = 6 }
			set_variable = { MAD_drought_migration_force = 0 }
			MAD_drought_migrate_worst_to_best = yes
		}
		else_if = {
			limit = {
				check_variable = { MAD_drought_severe_states < 0.5 }
				check_variable = { MAD_drought_depop > 0 }
			}
			subtract_from_variable = { MAD_drought_depop = 1 }
		}
		if = {
			limit = {
				check_variable = { MAD_drought_severe_states < 0.5 }
				check_variable = { MAD_drought_depop < 0.5 }
				has_idea = MAD_DROUGHT_EVACUATIONS
			}
			remove_ideas = MAD_DROUGHT_EVACUATIONS
		}

		MAD_drought_update_tier = yes
		# failed? (a year of exodus, net)
		if = {
			limit = {
				NOT = { has_country_flag = MAD_DROUGHT_FAILED_STATE }
				NOT = { has_country_flag = MAD_DROUGHT_CONTAINED_STATE }
				check_variable = { MAD_drought_depop > 11.99 }
			}
			set_country_flag = MAD_DROUGHT_FAILED_STATE
			country_event = { id = mad_drought.21 }
		}
		# over?
		if = {
			limit = { check_variable = { MAD_drought_severity < 0.001 } }
			MAD_drought_finish = yes
		}
	}
}

# State scope. Same floor rule as the national number: it cannot be talked away before the drought is decided.

	}
}

# A full consecutive-year containment qualification. The daily tick resets the counter to zero as soon as any
# requirement fails, so every one of the 365 days must remain valid. Once completed the flag/event is one-shot for
# that containment cycle; later deterioration clears the flag and allows another 365-day qualification.
MAD_drought_containment_daily_tick = {
	# A successful containment is not permanent: if the drought later breaks the containment conditions,
	# clear the cycle flag and make the player qualify for another full consecutive year.
	if = {
		limit = {
			has_country_flag = MAD_DROUGHT_CONTAINED_STATE
			NOT = { has_country_flag = MAD_DROUGHT_WON }
			NOT = { has_country_flag = MAD_DROUGHT_OVER }
			OR = {
				check_variable = { MAD_drought_prevention_count < 5 }
				check_variable = { MAD_drought_severity > 30.01 }
			}
		}
		clr_country_flag = MAD_DROUGHT_CONTAINED_STATE
		set_variable = { MAD_drought_containment_days = 0 }
	}

	if = {
		limit = {
			MAD_drought_exists = yes
			NOT = { has_country_flag = MAD_DROUGHT_CONTAINED_STATE }
			check_variable = { MAD_drought_prevention_count > 4.99 }
			check_variable = { MAD_drought_severity < 30.01 }
		}
		add_to_variable = { MAD_drought_containment_days = 1 }
		if = {
			limit = { check_variable = { MAD_drought_containment_days > 364.99 } }
			set_country_flag = MAD_DROUGHT_CONTAINED_STATE
			country_event = { id = mad_drought.22 }
		}
	}
	else = {
		if = {
			limit = { check_variable = { MAD_drought_containment_days > 0.001 } }
			set_variable = { MAD_drought_containment_days = 0 }
		}
	}

	# Keep the four stored map-frame variables synchronized daily as well as on monthly recalculation.
	# This makes existing saves update the mini-map immediately after the next daily tick instead of
	# waiting for the next 30-day drought tick.
	if = {
		limit = { MAD_drought_exists = yes }
		MAD_drought_sync_display = yes
	}
}

	if = { limit = { check_variable = { MAD_state_drought > 54.99 } } set_temp_variable = { MAD_st_tier = 3 } }
	if = { limit = { check_variable = { MAD_state_drought > 74.99 } } set_temp_variable = { MAD_st_tier = 4 } }

	if = {
		limit = { NOT = { check_variable = { MAD_state_drought_tier = MAD_st_tier } } }
		set_variable = { MAD_state_drought_tier = MAD_st_tier }
		MAD_drought_state_clear_tiers = yes
		if = { limit = { check_variable = { MAD_st_tier = 1 } } add_dynamic_modifier = { modifier = MAD_STATE_DROUGHT_1 } }
		if = { limit = { check_variable = { MAD_st_tier = 2 } } add_dynamic_modifier = { modifier = MAD_STATE_DROUGHT_2 } }
		if = { limit = { check_variable = { MAD_st_tier = 3 } } add_dynamic_modifier = { modifier = MAD_STATE_DROUGHT_3 } }
		if = { limit = { check_variable = { MAD_st_tier = 4 } } add_dynamic_modifier = { modifier = MAD_STATE_DROUGHT_4 } }
	}

	# the famine layer: hunger, not thirst. Comes and goes on its own meter.
	if = {
		limit = {
			check_variable = { MAD_state_famine > 39.99 }
			NOT = { has_dynamic_modifier = { modifier = MAD_STATE_FAMINE } }
		}
		add_dynamic_modifier = { modifier = MAD_STATE_FAMINE }
	}
	if = {
		limit = {
			check_variable = { MAD_state_famine < 40 }
			has_dynamic_modifier = { modifier = MAD_STATE_FAMINE }
		}
		remove_dynamic_modifier = { modifier = MAD_STATE_FAMINE }
	}
}

########################################################################
# THE PER-STATE MONTHLY STEP (called from MAD_drought_monthly_tick)
########################################################################
# Famine follows the drought with a lag: above 45 people start going hungry, below 45 they recover.
# A state that has had wells sunk (MAD_STATE_WELLS) recovers faster; one cut off from the network suffers more.

MAD_drought_state_monthly_one = {
	if = {
		limit = { check_variable = { MAD_state_relief_months > 0 } }
		subtract_from_variable = { MAD_state_relief_months = 1 }
		if = {
			limit = { check_variable = { MAD_state_relief_months < 0.001 } }
			add_to_variable = { MAD_state_drought = 4 }
			MAD_drought_state_clamp = yes
		}
	}
	set_temp_variable = { MAD_fam_t = MAD_state_drought }
	subtract_from_temp_variable = { MAD_fam_t = 45 }
	multiply_temp_variable = { MAD_fam_t = 0.12 }
	if = {
		limit = { has_state_flag = MAD_STATE_WELLS }
		multiply_temp_variable = { MAD_fam_t = 0.6 }
	}
	add_to_variable = { MAD_state_famine = MAD_fam_t }
	clamp_variable = { var = MAD_state_famine min = 0 max = 100 }
	MAD_drought_state_update_tier = yes
}

MAD_drought_state_monthly = {
	# Only the four modeled states get the local famine/water monthly step.

	# SUCCESS: both performance requirements are met before the three-year timer expires.
	if = {
		limit = {
			has_active_mission = MAD_JIRAMA_GENERATION_RECOVERY_MISSION
			NOT = { has_country_flag = MAD_JIRAMA_GENERATION_RECOVERY_EXPIRED }

		}

		limit = {
			has_country_flag = MAD_JIRAMA_GENERATION_RECOVERY_EXPIRED
			NOT = { has_completed_focus = MAD_JIRAMA_RECOVERY_SUCCEEDS }
			NOT = { has_completed_focus = MAD_JIRAMA_RECOVERY_FAILS }
		}

		complete_national_focus = MAD_JIRAMA_RECOVERY_FAILS
	}

		set_variable = { MAD_dahalo_receiving_investment_duration = 0.01 }
		set_variable = { MAD_dahalo_stability = -0.01 }
		set_variable = { MAD_dahalo_expected_police = 0.5 }
		set_variable = { MAD_dahalo_civilian_tax = 0 }
		set_variable = { MAD_dahalo_infrastructure_tax = 0 }
	}
	else = {
		set_variable = { MAD_dahalo_agri_productivity = 0 }
		set_variable = { MAD_dahalo_agri_income_tax = 0 }
		set_variable = { MAD_dahalo_agri_worker_req = 0 }
		set_variable = { MAD_dahalo_productivity_growth = 0 }
		set_variable = { MAD_dahalo_receiving_investment_cost = 0 }
		set_variable = { MAD_dahalo_receiving_investment_duration = 0 }
		set_variable = { MAD_dahalo_stability = 0 }
		set_variable = { MAD_dahalo_expected_police = 0 }
		set_variable = { MAD_dahalo_civilian_tax = 0 }
		set_variable = { MAD_dahalo_infrastructure_tax = 0 }
		# Reaching 0 threat is NOT the same as finishing the Dahalo storyline.
		# The terminal focus sets MAD_DAHALO_DEFEATED; this keeps the permanent panel and
		# the Remenabila hunt usable even if ordinary security work temporarily reaches 0.
		if = {
			limit = { has_dynamic_modifier = { modifier = MAD_dahalo_modifier } }
			remove_dynamic_modifier = { modifier = MAD_dahalo_modifier }
		}
	}

	if = {
		limit = {
			NOT = { has_country_flag = MAD_DAHALO_DEFEATED }
			NOT = { has_country_flag = MAD_REMENABILA_IN_POWER }
		}
		force_update_dynamic_modifier = yes
	}
}

MAD_dahalo_switch_to_remenabila = {
	set_variable = { MAD_dahalo_threat = 0 }
	MAD_dahalo_update_modifier = yes
}


MAD_dahalo_southern_supply_interference = {
	# This is deliberately limited to the Grand Sud and to physical relief/supply operations.
	# Cash transfers and locally-built wells are not penalised here: the problem is getting
	# water, food, seed and pest-control supplies safely into the countryside.
	#
	# Escort Relief Convoys temporarily flips the interception penalty off. This makes the
	# decision an actual countermeasure rather than a second generic threat-reduction button.
	if = {
		limit = {
			has_state_flag = MAD_GRAND_SUD_STATE
			NOT = { has_country_flag = MAD_DAHALO_DEFEATED }
			NOT = { has_country_flag = MAD_REMENABILA_IN_POWER }
			check_variable = { ROOT.MAD_dahalo_threat > 39.99 }
		}

		set_temp_variable = { MAD_dahalo_south_relief_factor = 0.80 }
		set_temp_variable = { MAD_dahalo_south_supply_leak = 1 }

		if = {
			limit = { has_country_flag = MAD_DAHALO_ESCORT_RELIEF_ACTIVE }
			set_temp_variable = { MAD_dahalo_south_relief_factor = 1.00 }
			set_temp_variable = { MAD_dahalo_south_supply_leak = 0 }
		}
		else_if = {
			limit = { check_variable = { ROOT.MAD_dahalo_threat > 79.99 } }
			set_temp_variable = { MAD_dahalo_south_relief_factor = 0.50 }
			set_temp_variable = { MAD_dahalo_south_supply_leak = 3 }
		}
		else_if = {
			limit = { check_variable = { ROOT.MAD_dahalo_threat > 59.99 } }
			set_temp_variable = { MAD_dahalo_south_relief_factor = 0.65 }
			set_temp_variable = { MAD_dahalo_south_supply_leak = 2 }
		}

		if = {
			limit = { check_variable = { MAD_dr_state_change < -0.001 } }
			multiply_temp_variable = { MAD_dr_state_change = MAD_dahalo_south_relief_factor }
		}
		if = {
			limit = { check_variable = { MAD_dr_state_famine < -0.001 } }
			multiply_temp_variable = { MAD_dr_state_famine = MAD_dahalo_south_relief_factor }
			add_to_variable = { MAD_state_famine = MAD_dahalo_south_supply_leak }
			clamp_variable = { var = MAD_state_famine min = 0 max = 100 }
		}
	}
}

MAD_dahalo_southern_water_relief_interference = {
	# Emergency water tankers lose part of their effective six-month relief window when
	# Dahalo bands are strong enough to divert or seize deliveries. An active escort
	# operation restores the full delivery window.
	if = {
		limit = {
			has_state_flag = MAD_GRAND_SUD_STATE
			NOT = { has_country_flag = MAD_DAHALO_DEFEATED }
			NOT = { has_country_flag = MAD_REMENABILA_IN_POWER }
			check_variable = { ROOT.MAD_dahalo_threat > 39.99 }
		}

		set_temp_variable = { MAD_dahalo_water_relief_months = 5 }
		if = {
			limit = { has_country_flag = MAD_DAHALO_ESCORT_RELIEF_ACTIVE }
			set_temp_variable = { MAD_dahalo_water_relief_months = 6 }
		}
		else_if = {
			limit = { check_variable = { ROOT.MAD_dahalo_threat > 79.99 } }
			set_temp_variable = { MAD_dahalo_water_relief_months = 3 }
		}
		else_if = {
			limit = { check_variable = { ROOT.MAD_dahalo_threat > 59.99 } }
			set_temp_variable = { MAD_dahalo_water_relief_months = 4 }
		}
		add_to_variable = { MAD_state_relief_months = MAD_dahalo_water_relief_months }
	}
	else = {
		add_to_variable = { MAD_state_relief_months = 6 }
	}
}

# Permit Arms is the accommodation route that can deliberately return the armed movement
# to maximum strength. Once the capped threat value is reached, the movement revolts openly
# in Toliara / the Grand Sud instead of remaining an ordinary insecurity penalty.

########################################################################
# RANDOM DAHALO INCIDENTS
########################################################################
# The incidents are deliberately infrequent and mostly one-shot. The repeatable ones represent
# the continuing insecurity of the cattle economy; the one-shot incidents create different long-term
# playthroughs by opening policy choices around local defence, private security and foreign investment.

MAD_dahalo_random_event_tick = {
	if = {
		limit = {
			check_variable = { MAD_dahalo_event_cooldown > 0.001 }
		}
		subtract_from_variable = { MAD_dahalo_event_cooldown = 1 }
	}
	else_if = {
		limit = {
			NOT = { has_country_flag = MAD_DAHALO_DEFEATED }
			NOT = { has_country_flag = MAD_REMENABILA_IN_POWER }
			NOT = { has_civil_war = yes }
			NOT = { has_country_flag = MAD_DAHALO_CIVIL_WAR_PENDING }
			NOT = { has_country_flag = MAD_REMENABILA_KILLED }
			NOT = { has_country_flag = MAD_DAHALO_SECURITY_ENDING }
			NOT = { has_country_flag = MAD_DAHALO_ACCOMMODATION_ENDING }
		}
		set_variable = { MAD_dahalo_event_cooldown = 180 }

		# Insecurity itself now worsens the famine in the Grand Sud, even if the player
		# does not click a relief decision. This is the background pressure that makes
		# high Dahalo threat strategically dangerous rather than merely inconvenient.
		if = {
			limit = {
				has_completed_focus = MAD_Drought_and_famine
			}
			268 = {
				if = {
					limit = { has_state_flag = MAD_GRAND_SUD_STATE }
					if = {
						limit = { check_variable = { ROOT.MAD_dahalo_threat > 79.99 } }
						add_to_variable = { MAD_state_famine = 3 }
					}
					else_if = {
						limit = { check_variable = { ROOT.MAD_dahalo_threat > 59.99 } }
						add_to_variable = { MAD_state_famine = 2 }
					}
					else_if = {
						limit = { check_variable = { ROOT.MAD_dahalo_threat > 39.99 } }
						add_to_variable = { MAD_state_famine = 1 }
					}
					clamp_variable = { var = MAD_state_famine min = 0 max = 100 }
					MAD_drought_state_update_tier = yes
				}
			}
		}
		random_list = {
			55 = { }
			17 = { country_event = { id = mad_dahalo.10 } }
			10 = { country_event = { id = mad_dahalo.11 } }
			8 = {
				if = {
					limit = {
						has_completed_focus = MAD_dahalo_military_ops
						check_variable = { MAD_dahalo_threat > 59.99 }
					}
					country_event = { id = mad_dahalo.15 }
				}
			}
			4 = {
				if = {
					limit = {
						NOT = { has_country_flag = MAD_DAHALO_LOCAL_DEFENCE_RESOLVED }
						check_variable = { MAD_dahalo_threat > 34.99 }
					}
					country_event = { id = mad_dahalo.12 }
				}
			}
			4 = {
				if = {
					limit = {
						date > 2003.1.1
						NOT = { has_country_flag = MAD_DAHALO_PRIVATE_SECURITY_RESOLVED }
						check_variable = { MAD_dahalo_threat > 44.99 }
					}
					country_event = { id = mad_dahalo.13 }
				}
			}
			2 = {
				if = {
					limit = {
						date > 2003.1.1
						NOT = { has_country_flag = MAD_DAHALO_PROJECT_SECURITY_RESOLVED }
						check_variable = { MAD_dahalo_threat > 59.99 }
						has_completed_focus = MAD_ACCEPT_FOREIGN_OWNERSHIP
						OR = {
							has_completed_focus = MAD_QMM_CONSTRUCTION
							has_completed_focus = MAD_Toliara_rare_earths
						}
					}
					country_event = { id = mad_dahalo.14 }
				}
			}
			3 = {
				if = {
					limit = {
						has_completed_focus = MAD_Hunt_for_red_with_a_swagger
						NOT = { has_country_flag = MAD_REMENABILA_LOCATED }
						NOT = { has_country_flag = MAD_REMENABILA_KILLED }
						NOT = { has_country_flag = MAD_DAHALO_REMENABILA_LEAD_RESOLVED }
						check_variable = { MAD_dahalo_threat > 29.99 }
					}
					country_event = { id = mad_dahalo.16 }
				}
			}
		}
	}
}

MAD_dahalo_daily_tick = {
	# The insecurity network slowly rebuilds itself after suppression: +2 threat per month (30-day equivalent).
	# Random incidents are additional shocks and remain possible until the system reaches a terminal ending.
	if = {
		limit = {
			NOT = { has_country_flag = MAD_DAHALO_DEFEATED }
			NOT = { has_country_flag = MAD_REMENABILA_IN_POWER }
			NOT = { has_country_flag = MAD_REMENABILA_KILLED }
			NOT = { has_country_flag = MAD_DAHALO_SECURITY_ENDING }
			NOT = { has_country_flag = MAD_DAHALO_ACCOMMODATION_ENDING }
			NOT = { has_country_flag = MAD_DAHALO_CIVIL_WAR_STARTED }
			NOT = { has_country_flag = MAD_DAHALO_CIVIL_WAR_PENDING }
			NOT = { has_civil_war = yes }
		}
		add_to_variable = { MAD_dahalo_threat = 0.0666667 }
		clamp_variable = { var = MAD_dahalo_threat min = 0 max = 100 }
		MAD_dahalo_update_modifier = yes
	}

	# Open revolt is checked after the passive pressure tick so the meter cannot be artificially held at 0.
	if = {
		limit = {
			has_completed_focus = MAD_PERMIT_ARMS
			NOT = { has_country_flag = MAD_DAHALO_CIVIL_WAR_STARTED }
			NOT = { has_country_flag = MAD_DAHALO_CIVIL_WAR_PENDING }
			NOT = { has_country_flag = MAD_REMENABILA_IN_POWER }
			NOT = { has_country_flag = MAD_REMENABILA_KILLED }
			NOT = { has_civil_war = yes }
			check_variable = { MAD_dahalo_threat > 99.99 }
		}
		set_country_flag = MAD_DAHALO_CIVIL_WAR_PENDING
		country_event = { id = mad_dahalo.1 }
	}
	MAD_dahalo_random_event_tick = yes
}

MAD_dahalo_prepare_for_civil_war = {
	set_country_flag = MAD_DAHALO_CIVIL_WAR_STARTED
	clr_country_flag = MAD_DAHALO_CIVIL_WAR_PENDING
	# Once this is an open civil war, stop applying the pre-war insecurity meter on top of it.
	set_variable = { MAD_dahalo_threat = 0 }
	MAD_dahalo_update_modifier = yes
}
