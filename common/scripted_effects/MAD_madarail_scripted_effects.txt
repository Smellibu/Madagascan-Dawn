# INVESTOR CYCLE
#
# Three suitors (Comazar, Vecturis, Holding), each seated once for 150 days. The player picks who comes
# next (event mad_madarail.3). The seated suitor's own concession decisions (MAD_MADARAIL_CONCEDE_*) raise
# MAD_madarail_confidence and are remembered as offers (flags MAD_MADARAIL_OFFER_1/2/3). If the investors commit,
# MAD_madarail_enact_concessions builds the company's dynamic modifier from the operator's base terms + the offers.
# Confidence >= 75 lets the player commit (STAY) by completing the mission MAD_MADARAIL_INVESTOR_TENURE.
########################################################################

# Who is seated is tracked with flags (no ideas). The flag stays set after the investors commit, so the category
# text keeps showing the operator; it is cleared when a suitor leaves or the investors desist.
MAD_madarail_clear_operators = {
	clr_country_flag = MAD_MADARAIL_SEATED_COMAZAR
	clr_country_flag = MAD_MADARAIL_SEATED_VECTURIS
	clr_country_flag = MAD_MADARAIL_SEATED_HOLDING
}

MAD_madarail_clear_concessions = {
	clr_country_flag = MAD_MADARAIL_OFFER_1
	clr_country_flag = MAD_MADARAIL_OFFER_2
	clr_country_flag = MAD_MADARAIL_OFFER_3
	# The Comazar freight penalty only applies while that committed concession remains active.
	remove_ideas = MAD_MADARAIL_COMAZAR_FREIGHT_CONCESSION
	MAD_rural_cooperation_calculate_monthly_change = yes
}

# Concessions are only OFFERS until the investors commit. This is where they become real, all in one go:
# the seated company's dynamic modifier is created from the operator's base terms plus one slice per offered
# concession (so which investor it is, and what was conceded, decide the quality), and the one-off costs are paid
# (treasury / ownership / stability). Before this runs nothing lasting exists, and a suitor who leaves without
# committing leaves nothing behind. Slots 1/2/3 match the MAD_MADARAIL_CONCEDE_* decisions.
# Variables: MAD_<company>_<modifier>, read by the dynamic modifiers in common/dynamic_modifiers.

# Seats a suitor: the previous one (and its concessions) leaves, confidence resets to the newcomer's starting
# level, the 150-day mission starts. Comazar is the hardest to win over, the Holding consortium the easiest.
MAD_madarail_seat_comazar = {
	MAD_madarail_clear_operators = yes
	MAD_madarail_clear_concessions = yes
	set_country_flag = MAD_MADARAIL_COMAZAR_USED
	set_country_flag = MAD_MADARAIL_SEATED_COMAZAR
	set_variable = { MAD_madarail_confidence = 10 }
	add_to_variable = { MAD_madarail_stage = 1 }
	activate_mission = MAD_MADARAIL_INVESTOR_TENURE
}

MAD_madarail_seat_vecturis = {
	MAD_madarail_clear_operators = yes
	MAD_madarail_clear_concessions = yes
	set_country_flag = MAD_MADARAIL_VECTURIS_USED
	set_country_flag = MAD_MADARAIL_SEATED_VECTURIS
	set_variable = { MAD_madarail_confidence = 30 }
	add_to_variable = { MAD_madarail_stage = 1 }

}

MAD_madarail_seat_holding = {

# Outcome 1: the investors commit. The suitor in place stays and every concession OFFERED now takes effect; STAY focus unlocks.
MAD_madarail_apply_stay_outcome = {
	MAD_madarail_enact_concessions = yes
	# Guaranteeing public freight for Comazar is the rural-distribution concession.
	# Its ongoing penalty starts only if Comazar commits after Freight Volumes was offered.
	if = {
		limit = {
			has_country_flag = MAD_MADARAIL_SEATED_COMAZAR
			has_country_flag = MAD_MADARAIL_OFFER_3
		}
		add_ideas = MAD_MADARAIL_COMAZAR_FREIGHT_CONCESSION
		custom_effect_tooltip = MAD_RC_MONTHLY_NEG_025_TT
	}
	set_country_flag = MAD_MADARAIL_OUTCOME_DECIDED
	set_country_flag = MAD_MADARAIL_STAY_UNLOCKED
	custom_effect_tooltip = MAD_madarail_stay_unlocked_tt
	MAD_rural_cooperation_calculate_monthly_change = yes
}

# Outcome 2: the investors leave (by the player's choice or because the third suitor ran out of time).
# Dire penalties start now; the Madarail state-ownership branch of the tree is what removes them.
MAD_madarail_apply_desist_outcome = {
	set_country_flag = MAD_MADARAIL_OUTCOME_DECIDED
	set_country_flag = MAD_MADARAIL_DESIST_UNLOCKED
	MAD_madarail_clear_operators = yes
	MAD_madarail_clear_concessions = yes
	if = {
		limit = { has_active_mission = MAD_MADARAIL_INVESTOR_TENURE }
		remove_mission = MAD_MADARAIL_INVESTOR_TENURE
	}
	add_ideas = MAD_MADARAIL_INVESTOR_EXODUS
	custom_effect_tooltip = MAD_madarail_desist_unlocked_tt
}

# Apply one Daewoo review delta and keep viability within its 0-100 range.
# Caller sets MAD_daewoo_viability_delta immediately before invoking this effect.
MAD_daewoo_adjust_viability = {
	add_to_variable = { MAD_daewoo_viability = MAD_daewoo_viability_delta }
	clamp_variable = { var = MAD_daewoo_viability min = 0 max = 100 }
}

# Puts the right lease tier on the country for the current MAD_daewoo_viability:
#   < 30  TOXIC      30 - 79.99  TROUBLED      80+  SOUND
# Only swaps when the tier actually changes.
MAD_daewoo_update_tier = {
	if = {
		limit = {
			check_variable = { MAD_daewoo_viability < 30 }
			NOT = { has_idea = MAD_DAEWOO_LEASE_TOXIC }
		}
		MAD_daewoo_clear_tiers = yes
		add_ideas = MAD_DAEWOO_LEASE_TOXIC
	}
	if = {
		limit = {
			check_variable = { MAD_daewoo_viability > 29.99 }
			check_variable = { MAD_daewoo_viability < 80 }
			NOT = { has_idea = MAD_DAEWOO_LEASE_TROUBLED }
		}
		MAD_daewoo_clear_tiers = yes
		add_ideas = MAD_DAEWOO_LEASE_TROUBLED
	}
	if = {
		limit = {
			check_variable = { MAD_daewoo_viability > 79.99 }
			NOT = { has_idea = MAD_DAEWOO_LEASE_SOUND }
		}
		MAD_daewoo_clear_tiers = yes
		add_ideas = MAD_DAEWOO_LEASE_SOUND
	}
}

MAD_daewoo_clear_tiers = {
	remove_ideas = MAD_DAEWOO_LEASE_TOXIC
	remove_ideas = MAD_DAEWOO_LEASE_TROUBLED
	remove_ideas = MAD_DAEWOO_LEASE_SOUND
}

# Shared lease-ending effects for collapse and managed sign-off.
# The cancellation focus keeps its outcome effects in the focus tree.

# The concession is gone, whichever way it went: its cash crops stop feeding the drought and every agricultural
# complex needs more hands again.
MAD_daewoo_break_effects = {
	if = {
		limit = { has_country_flag = MAD_DAEWOO_KOREAN_AGRI_SECTOR }
		hidden_effect = {
			add_to_variable = { MAD_agri_productivity = -0.05 }
			add_to_variable = { MAD_agri_income_tax = -0.02 }

			# The initial deal adds one district in state 268. Keeping the lease
			# adds another in 268 and one in 266. When the kept lease collapses,
			# remove both state-268 levels and the state-266 level.
			268 = {
				if = {
					limit = { agriculture_district > 1 }
					remove_building = {
						type = agriculture_district
						level = 2
					}
				}
				else = {
					if = {
						limit = { agriculture_district > 0 }
						remove_building = {
							type = agriculture_district
							level = 1
						}
					}
				}
			}
			if = {
				limit = { has_completed_focus = MAD_DEAL_KEPT }
				266 = {
					if = {
						limit = { agriculture_district > 0 }
						remove_building = {
							type = agriculture_district
							level = 1
						}
					}
				}
			}
			clr_country_flag = MAD_DAEWOO_KOREAN_AGRI_SECTOR
		}
	}
	set_variable = { MAD_drought_daewoo_pressure = 0 }
	custom_effect_tooltip = {
		localization_key = modifies_dynamic_modifier_tt
		MODIFIER = MAD_agriculture_modifier
	}
	add_to_variable = { MAD_agri_worker_req = 0.03 tooltip = agriculture_district_worker_requirement_modifier_tt }

}

# The lease ends: the tier idea goes (the Sound tier hands its farmland bonuses back in its on_remove) and the break effects run.
MAD_daewoo_end_lease = {
	MAD_daewoo_clear_tiers = yes
	MAD_daewoo_break_effects = yes
}

# These branches are consequences of the systems, not player choices.

MAD_autocomplete_forced_outcome_foci = {
	# The Daewoo lease has collapsed after being kept: force Deal Collapses.
	if = {
		limit = {
			has_completed_focus = MAD_DEAL_KEPT
			has_country_flag = MAD_DAEWOO_OUTCOME_FAILED
			NOT = { has_completed_focus = MAD_DEAL_KEPT_FAILED }
			NOT = { has_completed_focus = MAD_DEAL_KEPT_MANAGED_WELL }
		}
		complete_national_focus = MAD_DEAL_KEPT_FAILED
	}
}

# --- the drought hooks -------------------------------------------------------------------------------------------
# Korean cash crops feed the drought, but only while the drought exists. The monthly tick adds MAD_drought_daewoo_pressure
# to the severity every month (halved once the outgrower decision, flag MAD_DAEWOO_FOOD_QUOTA, is taken) and the whole
# term switches off the moment the drought is over.

MAD_daewoo_drought_kept = {
	if = {
		limit = { MAD_drought_exists = yes }
		custom_effect_tooltip = MAD_daewoo_drought_kept_tt
		add_to_variable = { MAD_drought_daewoo_pressure = 0.15 }
		set_temp_variable = { MAD_dr_change = 4 }
		MAD_drought_apply_change = yes
	}
}

# The success grows more cash crops on the extra estate, so it feeds the drought number harder.
MAD_daewoo_drought_well = {
	if = {
		limit = { MAD_drought_exists = yes }
		custom_effect_tooltip = MAD_daewoo_drought_well_tt
		# The initial kept-deal +0.15 is the total Daewoo pressure.
		# Successful management expands the estate but does not add more monthly pressure.
		set_temp_variable = { MAD_dr_change = 4 }
		MAD_drought_apply_change = yes
	}
}

# THE DROUGHT AND FAMINE PANEL
#
# One stat: MAD_drought_severity (0-100). Startup and the guarded MAD_Drought_and_famine focus initialise it;
# event mad_drought.2 calls MAD_drought_monthly_tick every 30 days until the drought is over.
#
# Every month:   severity += base pressure (creeps up every year)
#                          + Grand Sud term  (0.30, cut by up to 45% by infrastructure in the Toliara state, never to 0)
#                          + West term       (0.15, same rule, scored on state 266)
#                          + Daewoo cash crops (only while the drought exists; halved by the outgrower decision)
#                          + lingering shocks (historical droughts, failed rains; decays 20% a month)
#                          - preventative measures (permanent; each preventative decision adds to the total)
#                          - a recovery bonus once 5+ measures are in place
#                          - 2.5 more once Winning The Fight is done
#   Emergency water supply cuts a POSITIVE monthly rise to 30% for 6 months, then a +4 rebound.
#
# Phases:
#   MAD_DROUGHT_CONTAINED_STATE  set automatically: all 7 preventative measures and severity < 10.01 for one full year -> Winning The Fight opens
#   MAD_DROUGHT_WON              set by Winning The Fight
#   MAD_DROUGHT_OVER             severity reached 0 -> MAD_drought_resolved unlocks the further agricultural focuses
# Until containment or victory, severity cannot be pushed below 5; neglect leaves the drought active.

		limit = { check_variable = { MAD_drought_severity < 0.001 } }

# Scores infrastructure (0-3) in the Toliara state (268, the Grand Sud) into MAD_drought_infra_south.

# "$VAR$/$STATE$" effect (that syntax is Stellaris/Vic3/CK3-only and is not valid HOI4 script).
# State-scope infrastructure trigger: infrastructure 3 = score 1, 4 = score 2, 5 = score 3.
MAD_drought_score_south = {
	set_variable = { MAD_drought_infra_south = 0 }

	if = { limit = { 268 = { infrastructure > 3 } } add_to_variable = { MAD_drought_infra_south = 1 } }
	if = { limit = { 268 = { infrastructure > 4 } } add_to_variable = { MAD_drought_infra_south = 1 } }
}

# DROUGHT DISPLACEMENT

# The four modeled states use state_population_k (in thousands) to calculate population-proportional movement.
# Historical drought events alter drought/famine severity; departures arise only from this monthly severity check.
# Every qualifying calendar month contributes its own population fraction to that month's report:
#   state drought 60-79: 1/2000 of state population per qualifying month
#   state drought 80+:   1/500 of state population per qualifying month
# Fractions accumulate without rounding; total departures are rounded once when the monthly report is staged.
# A month below 60 contributes zero; each monthly report clears that month's accumulator after staging.
# add_manpower in STATE scope changes state population. A negative STATE add_manpower also creates a national
# recruitable-manpower side effect, so the same negative amount is applied to the country to cancel that side effect.
#
# To keep origin-state resettlement coherent, people already tracked as internally displaced are not selected for a
# second departure from their current host state. The amount is still calculated from total state population, then
# capped to that state's un-displaced resident population. This avoids counting the same displaced people twice.
# Counters ending in _idp_ORIGIN_DESTINATION contain living displaced people, in PEOPLE, currently in DESTINATION.

MAD_drought_displacement_snapshot = {
	266 = { set_variable = { ROOT.MAD_drought_mig_d_266 = MAD_state_drought } }
	267 = { set_variable = { ROOT.MAD_drought_mig_d_267 = MAD_state_drought } }
	268 = { set_variable = { ROOT.MAD_drought_mig_d_268 = MAD_state_drought } }
	487 = { set_variable = { ROOT.MAD_drought_mig_d_487 = MAD_state_drought } }
}

# Called once per calendar month in state scope. Accumulate that month's departures as PEOPLE,
# without rounding, so small monthly values are not lost. The monthly report rounds once when staged.
MAD_drought_state_displacement_tick = {
	if = {
		limit = { check_variable = { MAD_state_drought > 59.99 } }
		set_temp_variable = { MAD_disp_gross_people = state_population_k }
		multiply_temp_variable = { MAD_disp_gross_people = 1000 }
		if = {
			limit = { check_variable = { MAD_state_drought > 79.99 } }
			divide_temp_variable = { MAD_disp_gross_people = 500 }
		}
		else = {
			divide_temp_variable = { MAD_disp_gross_people = 2000 }
		}
		add_to_variable = { MAD_state_displacement_accumulator = MAD_disp_gross_people }
	}
}

# At the report boundary, turn each state's accumulated gross departures into the report slot for that origin.
# The source is capped at its resident population, excluding people already hosted as IDPs from earlier reports.
# Destination choice uses the latest four-state drought snapshot; the actual transfer waits for the report option.
MAD_drought_stage_displacement_period = {
	if = {
		limit = { check_variable = { MAD_state_displacement_accumulator > 0.5 } }
		set_temp_variable = { MAD_disp_gross_people = MAD_state_displacement_accumulator }
		round_temp_variable = MAD_disp_gross_people
		set_temp_variable = { MAD_disp_incoming_people = 0 }
		if = {
			limit = { check_variable = { ROOT.MAD_drought_transfer_source_id = 266 } }
			set_temp_variable = { MAD_disp_incoming_people = ROOT.MAD_drought_idp_267_266 }
			add_to_temp_variable = { MAD_disp_incoming_people = ROOT.MAD_drought_idp_268_266 }
			add_to_temp_variable = { MAD_disp_incoming_people = ROOT.MAD_drought_idp_487_266 }
		}
		else_if = {
			limit = { check_variable = { ROOT.MAD_drought_transfer_source_id = 267 } }
			set_temp_variable = { MAD_disp_incoming_people = ROOT.MAD_drought_idp_266_267 }
			add_to_temp_variable = { MAD_disp_incoming_people = ROOT.MAD_drought_idp_268_267 }
			add_to_temp_variable = { MAD_disp_incoming_people = ROOT.MAD_drought_idp_487_267 }
		}
		else_if = {
			limit = { check_variable = { ROOT.MAD_drought_transfer_source_id = 268 } }
			set_temp_variable = { MAD_disp_incoming_people = ROOT.MAD_drought_idp_266_268 }
			add_to_temp_variable = { MAD_disp_incoming_people = ROOT.MAD_drought_idp_267_268 }
			add_to_temp_variable = { MAD_disp_incoming_people = ROOT.MAD_drought_idp_487_268 }
		}
		else = {
			set_temp_variable = { MAD_disp_incoming_people = ROOT.MAD_drought_idp_266_487 }
			add_to_temp_variable = { MAD_disp_incoming_people = ROOT.MAD_drought_idp_267_487 }
			add_to_temp_variable = { MAD_disp_incoming_people = ROOT.MAD_drought_idp_268_487 }
		}
		set_temp_variable = { MAD_disp_resident_people = state_population_k }
		multiply_temp_variable = { MAD_disp_resident_people = 1000 }
		subtract_from_temp_variable = { MAD_disp_resident_people = MAD_disp_incoming_people }
		clamp_temp_variable = { var = MAD_disp_resident_people min = 0 }
		if = {
			limit = { check_variable = { MAD_disp_gross_people > MAD_disp_resident_people } }
			set_temp_variable = { MAD_disp_gross_people = MAD_disp_resident_people }
			round_temp_variable = MAD_disp_gross_people
		}
		if = {
			limit = { check_variable = { MAD_disp_gross_people > 0.5 } }
			if = {
				limit = { check_variable = { ROOT.MAD_drought_transfer_source_id = 266 } }
				save_event_target_as = MAD_drought_transfer_source_266
			}
			else_if = {
				limit = { check_variable = { ROOT.MAD_drought_transfer_source_id = 267 } }
				save_event_target_as = MAD_drought_transfer_source_267
			}
			else_if = {
				limit = { check_variable = { ROOT.MAD_drought_transfer_source_id = 268 } }
				save_event_target_as = MAD_drought_transfer_source_268
			}
			else = { save_event_target_as = MAD_drought_transfer_source_487 }
			set_variable = { ROOT.MAD_drought_transfer_gross_people = MAD_disp_gross_people }
			ROOT = { MAD_drought_execute_displacement = yes }
		}
	}
	set_variable = { MAD_state_displacement_accumulator = 0 }
}

MAD_drought_execute_displacement = {

		limit = { check_variable = { MAD_drought_transfer_gross_people > 0.5 } }
		set_temp_variable = { MAD_disp_survivors_people = MAD_drought_transfer_gross_people }
		multiply_temp_variable = { MAD_disp_survivors_people = 0.85 }
		round_temp_variable = MAD_disp_survivors_people
		set_temp_variable = { MAD_disp_deaths_people = MAD_drought_transfer_gross_people }
		subtract_from_temp_variable = { MAD_disp_deaths_people = MAD_disp_survivors_people }
		set_variable = { MAD_drought_transfer_survivors_people = MAD_disp_survivors_people }
		set_variable = { MAD_drought_transfer_deaths_people = MAD_disp_deaths_people }
		set_temp_variable = { MAD_disp_famine_relief = MAD_drought_transfer_gross_people }
		divide_temp_variable = { MAD_disp_famine_relief = 10000 }
		clamp_temp_variable = { var = MAD_disp_famine_relief min = 0.25 max = 2 }
		set_variable = { MAD_drought_transfer_famine_relief = MAD_disp_famine_relief }

		# Pick the destination from the four-state severity snapshot taken for this tick.
		set_variable = { MAD_drought_lowest_safe_id = 0 }
		set_variable = { MAD_drought_lowest_safe_severity = 101 }
		if = {
			limit = { NOT = { check_variable = { MAD_drought_transfer_source_id = 266 } } }
			if = { limit = { check_variable = { MAD_drought_mig_d_266 < 60 } check_variable = { MAD_drought_mig_d_266 < MAD_drought_lowest_safe_severity } } set_variable = { MAD_drought_lowest_safe_id = 266 } set_variable = { MAD_drought_lowest_safe_severity = MAD_drought_mig_d_266 } }
		}
		if = {
			limit = { NOT = { check_variable = { MAD_drought_transfer_source_id = 267 } } }
			if = { limit = { check_variable = { MAD_drought_mig_d_267 < 60 } check_variable = { MAD_drought_mig_d_267 < MAD_drought_lowest_safe_severity } } set_variable = { MAD_drought_lowest_safe_id = 267 } set_variable = { MAD_drought_lowest_safe_severity = MAD_drought_mig_d_267 } }
		}
		if = {
			limit = { NOT = { check_variable = { MAD_drought_transfer_source_id = 268 } } }
			if = { limit = { check_variable = { MAD_drought_mig_d_268 < 60 } check_variable = { MAD_drought_mig_d_268 < MAD_drought_lowest_safe_severity } } set_variable = { MAD_drought_lowest_safe_id = 268 } set_variable = { MAD_drought_lowest_safe_severity = MAD_drought_mig_d_268 } }
		}
		if = {
			limit = { NOT = { check_variable = { MAD_drought_transfer_source_id = 487 } } }

		set_variable = { MAD_drought_transfer_abroad_country = 0 }

			limit = { check_variable = { MAD_drought_lowest_safe_id > 0 } }
			set_variable = { MAD_drought_transfer_destination_id = MAD_drought_lowest_safe_id }
		}
		else = {
			# No other state is below 60. Prefer one of the established international destinations,
			# but never select a country that no longer owns any states.
			set_variable = { MAD_drought_transfer_abroad = 1 }
			random_list = { 34 = { set_variable = { MAD_drought_transfer_abroad_country = 1 } } 33 = { set_variable = { MAD_drought_transfer_abroad_country = 2 } } 33 = { set_variable = { MAD_drought_transfer_abroad_country = 3 } } }
			if = {
				limit = { check_variable = { MAD_drought_transfer_abroad_country = 1 } }
				if = {
					limit = { NOT = { country_exists = COM } }
					set_variable = { MAD_drought_transfer_abroad_country = 0 }
				}
				else_if = {
					limit = { NOT = { COM = { any_owned_state = { always = yes } } } }
					set_variable = { MAD_drought_transfer_abroad_country = 0 }
				}
			}
			if = {
				limit = { check_variable = { MAD_drought_transfer_abroad_country = 2 } }
				if = {
					limit = { NOT = { country_exists = MOZ } }
					set_variable = { MAD_drought_transfer_abroad_country = 0 }

				limit = { check_variable = { MAD_drought_transfer_abroad_country = 3 } }
				if = {
					limit = { NOT = { country_exists = SAF } }
					set_variable = { MAD_drought_transfer_abroad_country = 0 }
				}
				else_if = {
					limit = { NOT = { SAF = { any_owned_state = { always = yes } } } }
					set_variable = { MAD_drought_transfer_abroad_country = 0 }
				}
			}
			if = {
				limit = { check_variable = { MAD_drought_transfer_abroad_country < 0.5 } }
				# The random pick was unavailable. Try the preferred destinations in order.
				if = {
					limit = { country_exists = COM COM = { any_owned_state = { always = yes } } }
					set_variable = { MAD_drought_transfer_abroad_country = 1 }
				}
				else_if = {
					limit = { country_exists = MOZ MOZ = { any_owned_state = { always = yes } } }

					# use the least drought-affected other modeled state so population remains tracked.
					set_variable = { MAD_drought_lowest_safe_id = 0 }
					set_variable = { MAD_drought_lowest_safe_severity = 101 }

		# Save a destination scope under the unique report slot. Nothing is moved until the report option.
		if = {
			limit = { check_variable = { MAD_drought_transfer_abroad < 0.5 } }

			else_if = { limit = { check_variable = { MAD_drought_transfer_source_id = 487 } } 266 = { save_event_target_as = MAD_drought_transfer_destination_487 } }
		}
		else_if = {
			limit = { check_variable = { MAD_drought_transfer_destination_id = 267 } }
			if = { limit = { check_variable = { MAD_drought_transfer_source_id = 266 } } 267 = { save_event_target_as = MAD_drought_transfer_destination_266 } }
			else_if = { limit = { check_variable = { MAD_drought_transfer_source_id = 267 } } 267 = { save_event_target_as = MAD_drought_transfer_destination_267 } }

			else_if = { limit = { check_variable = { MAD_drought_transfer_source_id = 487 } } 267 = { save_event_target_as = MAD_drought_transfer_destination_487 } }
		}
		else_if = {
			limit = { check_variable = { MAD_drought_transfer_destination_id = 268 } }

			else_if = { limit = { check_variable = { MAD_drought_transfer_source_id = 267 } } 268 = { save_event_target_as = MAD_drought_transfer_destination_267 } }
			else_if = { limit = { check_variable = { MAD_drought_transfer_source_id = 268 } } 268 = { save_event_target_as = MAD_drought_transfer_destination_268 } }
			else_if = { limit = { check_variable = { MAD_drought_transfer_source_id = 487 } } 268 = { save_event_target_as = MAD_drought_transfer_destination_487 } }
		}
		else_if = {
			limit = { check_variable = { MAD_drought_transfer_destination_id = 487 } }
			if = { limit = { check_variable = { MAD_drought_transfer_source_id = 266 } } 487 = { save_event_target_as = MAD_drought_transfer_destination_266 } }
			else_if = { limit = { check_variable = { MAD_drought_transfer_source_id = 267 } } 487 = { save_event_target_as = MAD_drought_transfer_destination_267 } }
			else_if = { limit = { check_variable = { MAD_drought_transfer_source_id = 268 } } 487 = { save_event_target_as = MAD_drought_transfer_destination_268 } }
			else_if = { limit = { check_variable = { MAD_drought_transfer_source_id = 487 } } 487 = { save_event_target_as = MAD_drought_transfer_destination_487 } }
		}
		}
		else = {
		if = {
			limit = { check_variable = { MAD_drought_transfer_abroad_country = 1 } }
			if = { limit = { check_variable = { MAD_drought_transfer_source_id = 266 } } COM = { save_event_target_as = MAD_drought_transfer_destination_266 } }
			else_if = { limit = { check_variable = { MAD_drought_transfer_source_id = 267 } } COM = { save_event_target_as = MAD_drought_transfer_destination_267 } }
			else_if = { limit = { check_variable = { MAD_drought_transfer_source_id = 268 } } COM = { save_event_target_as = MAD_drought_transfer_destination_268 } }
			else_if = { limit = { check_variable = { MAD_drought_transfer_source_id = 487 } } COM = { save_event_target_as = MAD_drought_transfer_destination_487 } }
		}
		else_if = {
			limit = { check_variable = { MAD_drought_transfer_abroad_country = 2 } }
			if = { limit = { check_variable = { MAD_drought_transfer_source_id = 266 } } MOZ = { save_event_target_as = MAD_drought_transfer_destination_266 } }
			else_if = { limit = { check_variable = { MAD_drought_transfer_source_id = 267 } } MOZ = { save_event_target_as = MAD_drought_transfer_destination_267 } }

# Called once per calendar month by on_monthly_MAD.
# Each monthly tick collects that month's departure estimate, stages all four origins,
# and opens ONE report only when at least one origin has a positive staged departure.
# Empty months are silent: no event is queued and the regular transfer slots remain zero.
MAD_drought_displacement_monthly_tick = {
	MAD_drought_collect_displacement_month = yes
	if = {
		limit = { NOT = { has_country_flag = MAD_DROUGHT_DISPLACEMENT_REPORT_PENDING } }
		MAD_drought_displacement_snapshot = yes
		MAD_drought_clear_regular_transfer_slots = yes
		set_variable = { MAD_drought_transfer_source_id = 266 }
		266 = { MAD_drought_stage_displacement_period = yes }

		267 = { MAD_drought_stage_displacement_period = yes }
		set_variable = { MAD_drought_transfer_source_id = 268 }
		268 = { MAD_drought_stage_displacement_period = yes }
		set_variable = { MAD_drought_transfer_source_id = 487 }
		487 = { MAD_drought_stage_displacement_period = yes }
		MAD_drought_prepare_displacement_report_summary = yes
		if = {
			limit = {
				OR = {
					check_variable = { MAD_drought_transfer_gross_266 > 0.5 }
					check_variable = { MAD_drought_transfer_gross_267 > 0.5 }
					check_variable = { MAD_drought_transfer_gross_268 > 0.5 }
					check_variable = { MAD_drought_transfer_gross_487 > 0.5 }
				}
			}
			set_country_flag = MAD_DROUGHT_DISPLACEMENT_REPORT_PENDING
			country_event = { id = mad_drought.23 hours = 1 }
		}
	}
}

# Internal resettlement reverses tracked state-to-state transfers, up to 20,000 people per use.
# The expat programme returns 10% of the original living diaspora (those who survived the journey), limited to six calls.

MAD_drought_return_route_266_267 = {
	if = {
		limit = {
			check_variable = { MAD_drought_resettle_remaining > 0.5 }
			check_variable = { MAD_drought_idp_266_267 > 0.5 }
		}
		set_temp_variable = { MAD_drought_return_people = MAD_drought_idp_266_267 }
		if = {
			limit = { check_variable = { MAD_drought_return_people > MAD_drought_resettle_remaining } }
			set_temp_variable = { MAD_drought_return_people = MAD_drought_resettle_remaining }
		}
		round_temp_variable = MAD_drought_return_people
		if = {
			limit = { check_variable = { MAD_drought_return_people > 0.5 } }
			set_variable = { MAD_drought_return_people_shared = MAD_drought_return_people }
			set_variable = { MAD_drought_return_negative_people_shared = MAD_drought_return_people_shared }
			multiply_variable = { MAD_drought_return_negative_people_shared = -1 }
			266 = { set_temp_variable = { MAD_disp_state_pop_change = ROOT.MAD_drought_return_people_shared } add_manpower = var:MAD_disp_state_pop_change }
			267 = { set_temp_variable = { MAD_disp_state_pop_change = ROOT.MAD_drought_return_negative_people_shared } add_manpower = var:MAD_disp_state_pop_change }
			# Cancel the recruitable-manpower side effect of the negative state-scope add_manpower.
			add_manpower = var:MAD_drought_return_negative_people_shared
			subtract_from_variable = { MAD_drought_idp_266_267 = MAD_drought_return_people }
			subtract_from_variable = { MAD_drought_displaced_internal = MAD_drought_return_people }
			subtract_from_variable = { MAD_drought_displaced = MAD_drought_return_people }
			subtract_from_variable = { MAD_drought_resettle_remaining = MAD_drought_return_people }
			clamp_variable = { var = MAD_drought_idp_266_267 min = 0 }
		}
	}
}

MAD_drought_monthly_tick = {
	if = {
		limit = { MAD_drought_exists = yes }

		# the climate creeps: every twelve months the base pressure rises (until it is won)
		add_to_variable = { MAD_drought_months = 1 }
		add_to_variable = { MAD_drought_calendar_months = 1 }
		if = {
			limit = { check_variable = { MAD_drought_months > 11.99 } }
			set_variable = { MAD_drought_months = 0 }
			if = {
				limit = {
					check_variable = { MAD_drought_base_pressure < 0.65 }
					NOT = { has_country_flag = MAD_DROUGHT_WON }
				}
				add_to_variable = { MAD_drought_base_pressure = 0.03 }
			}
		}

		# history and chance
		MAD_drought_scheduled_events = yes

		# gross pressure this month
		set_temp_variable = { MAD_dr_delta = MAD_drought_base_pressure }
		MAD_drought_read_infrastructure = yes
		MAD_drought_add_south_term = yes
		MAD_drought_add_west_term = yes

		# Daewoo's cash crops - only while the drought exists (this whole block is inside that check)
		if = {
			limit = { check_variable = { MAD_drought_daewoo_pressure > 0 } }
			set_temp_variable = { MAD_dr_term = MAD_drought_daewoo_pressure }
			if = {
				limit = { has_country_flag = MAD_DAEWOO_FOOD_QUOTA }
				multiply_temp_variable = { MAD_dr_term = 0.5 }
			}
			add_to_temp_variable = { MAD_dr_delta = MAD_dr_term }
		}

		# lingering shocks
		add_to_temp_variable = { MAD_dr_delta = MAD_drought_shock_pressure }
		multiply_variable = { MAD_drought_shock_pressure = 0.8 }

		# what has been built to prevent it
		subtract_from_temp_variable = { MAD_dr_delta = MAD_drought_prevention_total }
		subtract_from_temp_variable = { MAD_dr_delta = MAD_drought_cna_pressure_reduction }
		# Migration guard for older saves where CNA work had been incorrectly counted as 10/7 formal measures.
		clamp_variable = { var = MAD_drought_prevention_count min = 0 max = 7 }
		if = {
			limit = { check_variable = { MAD_drought_prevention_count > 4.99 } }
			set_temp_variable = { MAD_dr_term = MAD_drought_prevention_count }
			subtract_from_temp_variable = { MAD_dr_term = 4 }
			multiply_temp_variable = { MAD_dr_term = 0.2 }
			subtract_from_temp_variable = { MAD_dr_delta = MAD_dr_term }
		}
		if = {
			limit = { has_country_flag = MAD_DROUGHT_WON }
			subtract_from_temp_variable = { MAD_dr_delta = 2.5 }
		}

		# apply it - through the states, which then set the national average
		set_variable = { MAD_drought_last_delta = MAD_dr_delta }
		set_temp_variable = { MAD_dr_change = MAD_dr_delta }
		MAD_drought_apply_change = yes
		# each state then drifts a little on its own and its famine meter follows the drought
		MAD_drought_state_monthly = yes
		# State-monthly relief expiry/drift changes MAD_state_drought after MAD_drought_apply_change
		# has already recalculated the average. Recalculate again so MAD_drought_severity and the map stay current.
		MAD_drought_recalc_national = yes

		# People leave the states that are actually in severe drought, not just because the national average is high.
		# The severe-state counter keeps the evacuation response active while the drought remains severe; actual people are moved below.
		set_variable = { MAD_drought_severe_states = 0 }
		266 = { if = { limit = { check_variable = { MAD_state_drought > 59.99 } } ROOT = { add_to_variable = { MAD_drought_severe_states = 1 } } } }
		267 = { if = { limit = { check_variable = { MAD_state_drought > 59.99 } } ROOT = { add_to_variable = { MAD_drought_severe_states = 1 } } } }
		268 = { if = { limit = { check_variable = { MAD_state_drought > 59.99 } } ROOT = { add_to_variable = { MAD_drought_severe_states = 1 } } } }
		487 = { if = { limit = { check_variable = { MAD_state_drought > 59.99 } } ROOT = { add_to_variable = { MAD_drought_severe_states = 1 } } } }
		if = {
			limit = { check_variable = { MAD_drought_severe_states > 0.5 } }
			add_to_variable = { MAD_drought_depop = 1 }
			if = {
				limit = { NOT = { has_idea = MAD_DROUGHT_EVACUATIONS } }
				add_ideas = MAD_DROUGHT_EVACUATIONS
			}

		}
		else_if = {
			limit = {
				check_variable = { MAD_drought_severe_states < 0.5 }
				check_variable = { MAD_drought_depop > 0 }
			}
			subtract_from_variable = { MAD_drought_depop = 1 }
		}
		if = {
			limit = {
				check_variable = { MAD_drought_severe_states < 0.5 }
				check_variable = { MAD_drought_depop < 0.5 }
				has_idea = MAD_DROUGHT_EVACUATIONS
			}
			remove_ideas = MAD_DROUGHT_EVACUATIONS
		}

		# Regular displacement is handled by the separate calendar-month on_action.
		# Keep this 30-day climate/severity tick separate from the monthly displacement report tick.

		# Finish as soon as severity reaches zero.
		if = {
			limit = { check_variable = { MAD_drought_severity < 0.001 } }
			MAD_drought_finish = yes
		}
	}
}

# A full consecutive-year containment qualification. The daily tick resets the counter to zero as soon as any
# requirement fails, so every one of the 365 days must remain valid. Once completed the flag/event is one-shot for
# that containment cycle; later deterioration clears the flag and allows another 365-day qualification.
MAD_drought_containment_daily_tick = {
	# A successful containment is not permanent: if the drought later breaks the containment conditions,
	# clear the cycle flag and make the player qualify for another full consecutive year.
	if = {
		limit = {

			}
		}
		clr_country_flag = MAD_DROUGHT_CONTAINED_STATE
		set_variable = { MAD_drought_containment_days = 0 }
	}

	if = {
		limit = {
			MAD_drought_exists = yes
			NOT = { has_country_flag = MAD_DROUGHT_CONTAINED_STATE }
			check_variable = { MAD_drought_prevention_count > 6.99 }
			check_variable = { MAD_drought_severity < 10.01 }
		}
		add_to_variable = { MAD_drought_containment_days = 1 }
		if = {
			limit = { check_variable = { MAD_drought_containment_days > 364.99 } }
			set_country_flag = MAD_DROUGHT_CONTAINED_STATE
			country_event = mad_drought.22
		}
	}
	else = {
		if = {
			limit = { check_variable = { MAD_drought_containment_days > 0.001 } }
			set_variable = { MAD_drought_containment_days = 0 }
		}
	}

	# Keep the four stored map-frame variables synchronized daily as well as on monthly recalculation.
	# This makes existing saves update the mini-map immediately after the next daily tick instead of
	# waiting for the next 30-day drought tick.
	if = {
		limit = { MAD_drought_exists = yes }
		MAD_drought_sync_display = yes
	}

########################################################################
# THE PER-STATE MONTHLY STEP (called from MAD_drought_monthly_tick)
########################################################################
# Famine follows the drought with a lag: above 45 the meter rises, below 45 it recovers.
# Wells (MAD_STATE_WELLS) reduce famine growth above 45 and accelerate recovery below 45.

MAD_drought_state_monthly_one = {
	if = {
		limit = { check_variable = { MAD_state_relief_months > 0 } }
		subtract_from_variable = { MAD_state_relief_months = 1 }
		if = {
			limit = { check_variable = { MAD_state_relief_months < 0.001 } }
			add_to_variable = { MAD_state_drought = 4 }
			MAD_drought_state_clamp = yes
		}
	}
	set_temp_variable = { MAD_fam_t = MAD_state_drought }
	subtract_from_temp_variable = { MAD_fam_t = 45 }
	multiply_temp_variable = { MAD_fam_t = 0.12 }
	if = {
		limit = {
			has_state_flag = MAD_STATE_WELLS
			check_variable = { MAD_fam_t > 0 }
		}
		multiply_temp_variable = { MAD_fam_t = 0.6 }
	}
	if = {
		limit = {
			has_state_flag = MAD_STATE_WELLS
			check_variable = { MAD_fam_t < 0 }
		}
		multiply_temp_variable = { MAD_fam_t = 1.4 }
	}
	add_to_variable = { MAD_state_famine = MAD_fam_t }
	clamp_variable = { var = MAD_state_famine min = 0 max = 100 }
	MAD_drought_state_update_tier = yes
}

########################################################################
# DAILY AUTO-RESOLUTION
# Called once per day for Madagascar from on_daily_MAD.
########################################################################

MAD_JIRAMA_GENERATION_RECOVERY_DAILY = {

	# Also cover saves where the recovery flag was set before this visible mission existed.
	if = {
		limit = {
			has_country_flag = MAD_JIRAMA_GENERATION_RECOVERY_STARTED
			NOT = { has_country_flag = MAD_JIRAMA_GENERATION_RECOVERY_EXPIRED }
			NOT = { has_country_flag = MAD_JIRAMA_GENERATION_RECOVERY_REQUIREMENTS_MET }
			NOT = { has_completed_focus = MAD_JIRAMA_RECOVERY_SUCCEEDS }
			NOT = { has_completed_focus = MAD_JIRAMA_RECOVERY_FAILS }
			NOT = { has_active_mission = MAD_JIRAMA_GENERATION_RECOVERY_MISSION }
		}
		activate_mission = MAD_JIRAMA_GENERATION_RECOVERY_MISSION
	}

	# MD's energy_calc_balance computes this BEFORE storage discharge and fossil
	# curtailment adjustments:
	#   energy_balance = energy_sum - energy_consumption
	# energy_sum contains all generation sources (renewable, hydro, geothermal,
	# fossil, nuclear and other energy gain). Therefore generation >= 140% of use
	# is exactly equivalent to an energy balance >= 40% of consumption. This avoids
	# source-by-source reconstruction and does not mistake stored energy for generation.
	if = {
		limit = {
			has_country_flag = MAD_JIRAMA_GENERATION_RECOVERY_STARTED
			NOT = { has_completed_focus = MAD_JIRAMA_RECOVERY_SUCCEEDS }
			NOT = { has_completed_focus = MAD_JIRAMA_RECOVERY_FAILS }
			has_variable = energy_balance
			has_variable = energy_consumption
		}
		set_variable = { MAD_JIRAMA_required_surplus = energy_consumption }
		multiply_variable = { MAD_JIRAMA_required_surplus = 0.40 }
	}

	# The visible mission owns the 1095-day deadline and sets the expired flag.

	# SUCCESS: requirements must be met before the three-year timer expires.
	if = {
		limit = {
			NOT = { has_country_flag = MAD_JIRAMA_GENERATION_RECOVERY_EXPIRED }
			NOT = { has_completed_focus = MAD_JIRAMA_RECOVERY_SUCCEEDS }
			NOT = { has_completed_focus = MAD_JIRAMA_RECOVERY_FAILS }
			MAD_JIRAMA_RECOVERY_REQUIREMENTS_MET = yes
		}
		log = "[GetDateText]: [This.GetName]: JIRAMA recovery requirements met"
		set_country_flag = MAD_JIRAMA_GENERATION_RECOVERY_REQUIREMENTS_MET
		remove_mission = MAD_JIRAMA_GENERATION_RECOVERY_MISSION
	}
}

MAD_dahalo_southern_supply_interference = {
	# This is deliberately limited to the Grand Sud and to physical relief/supply operations.
	# Cash transfers and locally-built wells are not penalised here: the problem is getting
	# water, food, seed and pest-control supplies safely into the countryside.
	#
	# Escort Relief Convoys temporarily flips the interception penalty off. This makes the
	# decision an actual countermeasure rather than a second generic threat-reduction button.
	if = {
		limit = {
			has_state_flag = MAD_GRAND_SUD_STATE
			NOT = {
				ROOT = {
					has_country_flag = MAD_DAHALO_DEFEATED
				}
			}
			NOT = {
				ROOT = {
					has_country_flag = MAD_REMENABILA_IN_POWER
				}
			}
			check_variable = { ROOT.MAD_dahalo_threat > 39.99 }
		}

		set_temp_variable = { MAD_dahalo_south_relief_factor = 0.80 }
		set_temp_variable = { MAD_dahalo_south_supply_leak = 1 }

		if = {
			limit = {
				ROOT = {
					has_country_flag = MAD_DAHALO_ESCORT_RELIEF_ACTIVE
				}
			}
			set_temp_variable = { MAD_dahalo_south_relief_factor = 1.00 }
			set_temp_variable = { MAD_dahalo_south_supply_leak = 0 }
		}
		else_if = {
			limit = { check_variable = { ROOT.MAD_dahalo_threat > 79.99 } }
			set_temp_variable = { MAD_dahalo_south_relief_factor = 0.50 }
			set_temp_variable = { MAD_dahalo_south_supply_leak = 3 }
		}
		else_if = {
			limit = { check_variable = { ROOT.MAD_dahalo_threat > 59.99 } }
			set_temp_variable = { MAD_dahalo_south_relief_factor = 0.65 }
			set_temp_variable = { MAD_dahalo_south_supply_leak = 2 }
		}

		if = {
			limit = { check_variable = { MAD_dr_state_change < -0.001 } }
			multiply_temp_variable = { MAD_dr_state_change = MAD_dahalo_south_relief_factor }
		}
		if = {
			limit = { check_variable = { MAD_dr_state_famine < -0.001 } }
			multiply_temp_variable = { MAD_dr_state_famine = MAD_dahalo_south_relief_factor }
			add_to_variable = { MAD_state_famine = MAD_dahalo_south_supply_leak }
			clamp_variable = { var = MAD_state_famine min = 0 max = 100 }
		}
	}
}

########################################################################
# DAHALO MAX-STRENGTH / CIVIL WAR CHECK
########################################################################
# Permit Arms is the accommodation route that can deliberately return the armed movement
# to maximum strength. Once the capped threat value is reached, the movement revolts openly
# in Toliara / the Grand Sud instead of remaining an ordinary insecurity penalty.

########################################################################
# RANDOM DAHALO INCIDENTS
########################################################################
# The incidents are deliberately infrequent and mostly one-shot. The repeatable ones represent
# the continuing insecurity of the cattle economy; the one-shot incidents create different long-term
# playthroughs by opening policy choices around local defence, private security and foreign investment.

MAD_dahalo_random_event_tick = {
	if = {
		limit = {
			check_variable = { MAD_dahalo_event_cooldown > 0.001 }
		}
		subtract_from_variable = { MAD_dahalo_event_cooldown = 1 }
	}
	if = {
		limit = {
			check_variable = { MAD_dahalo_event_cooldown < 0.001 }
			NOT = { has_country_flag = MAD_DAHALO_DEFEATED }
			NOT = { has_country_flag = MAD_REMENABILA_IN_POWER }
			NOT = { has_civil_war = yes }
			NOT = { has_country_flag = MAD_DAHALO_CIVIL_WAR_PENDING }
			NOT = { has_country_flag = MAD_REMENABILA_KILLED }
			NOT = { has_country_flag = MAD_DAHALO_SECURITY_ENDING }
			NOT = { has_country_flag = MAD_DAHALO_ACCOMMODATION_ENDING }
		}
		set_variable = { MAD_dahalo_event_cooldown = 180 }

		# Insecurity itself now worsens the famine in the Grand Sud, even if the player
		# does not click a relief decision. This is the background pressure that makes
		# high Dahalo threat strategically dangerous rather than merely inconvenient.
		if = {
			limit = {
				has_completed_focus = MAD_Drought_and_famine
			}
			268 = {
				if = {
					limit = { has_state_flag = MAD_GRAND_SUD_STATE }
					if = {
						limit = { check_variable = { ROOT.MAD_dahalo_threat > 79.99 } }
						add_to_variable = { MAD_state_famine = 3 }
					}
					else_if = {
						limit = { check_variable = { ROOT.MAD_dahalo_threat > 59.99 } }
						add_to_variable = { MAD_state_famine = 2 }
					}
					else_if = {
						limit = { check_variable = { ROOT.MAD_dahalo_threat > 39.99 } }
						add_to_variable = { MAD_state_famine = 1 }
					}
					clamp_variable = { var = MAD_state_famine min = 0 max = 100 }
					MAD_drought_state_update_tier = yes
				}
			}
		}
		random_list = {
			55 = { }
			17 = { country_event = mad_dahalo.10 }
			10 = { country_event = mad_dahalo.11 }
			8 = {
				if = {
					limit = {
						has_completed_focus = MAD_dahalo_military_ops
						check_variable = { MAD_dahalo_threat > 59.99 }
					}
					country_event = mad_dahalo.15
				}
			}
			4 = {
				if = {
					limit = {
						NOT = { has_country_flag = MAD_DAHALO_LOCAL_DEFENCE_RESOLVED }
						check_variable = { MAD_dahalo_threat > 34.99 }
					}
					country_event = mad_dahalo.12
				}
			}
			4 = {
				if = {
					limit = {
						date > 2003.1.1
						NOT = { has_country_flag = MAD_DAHALO_PRIVATE_SECURITY_RESOLVED }
						check_variable = { MAD_dahalo_threat > 44.99 }
					}
					country_event = mad_dahalo.13
				}
			}
			2 = {
				if = {
					limit = {
						date > 2003.1.1
						NOT = { has_country_flag = MAD_DAHALO_PROJECT_SECURITY_RESOLVED }
						check_variable = { MAD_dahalo_threat > 59.99 }
						has_completed_focus = MAD_ACCEPT_FOREIGN_OWNERSHIP
						OR = {
							has_completed_focus = MAD_QMM_CONSTRUCTION
							has_completed_focus = MAD_Toliara_rare_earths
						}
					}
					country_event = mad_dahalo.14
				}
			}
			3 = {
				if = {
					limit = {
						has_completed_focus = MAD_Hunt_for_red_with_a_swagger
						NOT = { has_country_flag = MAD_REMENABILA_LOCATED }
						NOT = { has_country_flag = MAD_REMENABILA_KILLED }
						NOT = { has_country_flag = MAD_DAHALO_REMENABILA_LEAD_RESOLVED }
						check_variable = { MAD_dahalo_threat > 29.99 }
					}
					country_event = mad_dahalo.16
				}
			}
		}
	}
}

MAD_dahalo_daily_tick = {
	# Daily incident processing only; passive Dahalo threat growth is handled by MAD_dahalo_monthly_tick.
	# Random incidents are additional shocks and remain possible until the system reaches a terminal ending.
	if = {
		limit = {
			NOT = { has_country_flag = MAD_DAHALO_DEFEATED }
			NOT = { has_country_flag = MAD_REMENABILA_IN_POWER }
			NOT = { has_country_flag = MAD_REMENABILA_KILLED }
			NOT = { has_country_flag = MAD_DAHALO_SECURITY_ENDING }
			NOT = { has_country_flag = MAD_DAHALO_ACCOMMODATION_ENDING }
			NOT = { has_country_flag = MAD_DAHALO_CIVIL_WAR_STARTED }
			NOT = { has_country_flag = MAD_DAHALO_CIVIL_WAR_PENDING }
			NOT = { has_civil_war = yes }
		}

		clamp_variable = { var = MAD_dahalo_threat min = 0 max = 100 }
		MAD_dahalo_update_modifier = yes
	}

	# Open revolt is checked during daily processing so the meter cannot be artificially held at 0.
	if = {
		limit = {
			has_completed_focus = MAD_PERMIT_ARMS
			NOT = { has_country_flag = MAD_DAHALO_CIVIL_WAR_STARTED }
			NOT = { has_country_flag = MAD_DAHALO_CIVIL_WAR_PENDING }
			NOT = { has_country_flag = MAD_REMENABILA_IN_POWER }
			NOT = { has_country_flag = MAD_REMENABILA_KILLED }
			NOT = { has_civil_war = yes }
			check_variable = { MAD_dahalo_threat > 99.99 }
		}
		set_country_flag = MAD_DAHALO_CIVIL_WAR_PENDING
		country_event = mad_dahalo.1
	}
	MAD_dahalo_random_event_tick = yes
}

					has_country_flag = MAD_AREMA_RENEWAL
					has_country_flag = MAD_AREMA_RECONCILIATION

				limit = {
					check_variable = { MAD_RATSIRAKA_FORMATION_STABLE_DAYS > 2 }

						check_variable = { var = party_pop_elect_array^ruling_party value = majority_threshold compare = greater_than_or_equals }

						has_elections = no
					}
				}
				set_country_flag = MAD_RATSIRAKA_EVENT_QUEUED
				country_event = { id = mad_political.90 days = 1 }
			}
		}
	}
}

MAD_rural_cooperation_calculate_monthly_change = {
	# Recalculate the cap-aware monthly rate cached for the category description.
	# This only updates the displayed/cached rate; it does not apply a monthly meter change.
	# Source scripts call this immediately after changing an input to the monthly-rate calculation.
	# There is no election, government-transition, or daily polling hook.
	set_temp_variable = { MAD_rc_monthly_gain = 0.50 }
	set_variable = { MAD_rural_cooperation_cap_adjustment = 0 }
	set_variable = { MAD_rural_cooperation_meter_upper_adjustment = 0 }
	set_variable = { MAD_rural_cooperation_meter_lower_adjustment = 0 }
	# Constitutional secularism dampens rural cooperation; open state-church cooperation strengthens it.
	if = { limit = { has_idea = MAD_POL_SECULAR_GOVERNMENT } add_to_temp_variable = { MAD_rc_monthly_gain = -0.50 } }
	if = { limit = { has_idea = MAD_POL_UNSECULAR_GOVERNMENT } add_to_temp_variable = { MAD_rc_monthly_gain = 0.50 } }
	if = { limit = { has_idea = MAD_POL_ANDRIANA_HERITAGE_RECOGNISED } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { has_idea = MAD_POL_ANDRIANA_NON_RECOGNITION } add_to_temp_variable = { MAD_rc_monthly_gain = -0.25 } }
	# MD stores inflation_rate_var as a fraction: 0.25 = 25%, 0.50 = 50%, 0.75 = 75%.
	if = {
		limit = {
			has_variable = inflation_rate_var
			NOT = { check_variable = { inflation_rate_var < 0.75 } }
		}
		add_to_temp_variable = { MAD_rc_monthly_gain = -1.00 }
	}
	else_if = {
		limit = {
			has_variable = inflation_rate_var
			NOT = { check_variable = { inflation_rate_var < 0.50 } }
		}
		add_to_temp_variable = { MAD_rc_monthly_gain = -0.50 }
	}
	else_if = {
		limit = {
			has_variable = inflation_rate_var
			NOT = { check_variable = { inflation_rate_var < 0.25 } }
		}
		add_to_temp_variable = { MAD_rc_monthly_gain = -0.25 }
	}

	# Trust follows basic state capacity, public consent, and a countryside not in crisis.
	# Stability tiers deliberately stack: above 65%, both +0.25 effects apply (= +0.50 total).
	# Scripted localisation hides the lower-tier row when the cumulative higher tier is active.
	if = { limit = { has_stability > 0.499 } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { has_stability > 0.649 } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }

	# Government popularity is checked against the ruling ideology group rather than a fixed party.
	# Popularity tiers deliberately stack: above 60%, both +0.25 effects apply (= +0.50 total).
	# Scripted localisation hides the lower-tier row when the cumulative higher tier is active.
	if = {
		limit = {
			OR = {
			AND = { has_government = democratic democratic > 0.499 }
			AND = { has_government = communism communism > 0.499 }
			AND = { has_government = fascism fascism > 0.499 }
			AND = { has_government = neutrality neutrality > 0.499 }
			}
		}
		add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 }
	}
	if = {
		limit = {
			OR = {
			AND = { has_government = democratic democratic > 0.599 }
			AND = { has_government = communism communism > 0.599 }
			AND = { has_government = fascism fascism > 0.599 }
			AND = { has_government = neutrality neutrality > 0.599 }
			}
		}
		add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 }
	}

	if = { limit = { check_variable = { MAD_drought_severity < 50 } check_variable = { MAD_drought_severity > 20 } } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { NOT = { check_variable = { MAD_drought_severity > 20 } } } add_to_temp_variable = { MAD_rc_monthly_gain = 0.5 } }
	if = { limit = { check_variable = { MAD_dahalo_threat < 50 } check_variable = { MAD_dahalo_threat > 25 }  } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { NOT = { check_variable = { MAD_dahalo_threat > 25 } } } add_to_temp_variable = { MAD_rc_monthly_gain = 0.50 } }

	# Governance paths provide lasting institutional memory. Milestones stack, subject to the +3 cap.
	if = { limit = { has_completed_focus = MAD_POL_ROY_CROWN_WITHOUT_KING } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { has_completed_focus = MAD_POL_ROY_FOKONOLONA } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { has_completed_focus = MAD_POL_ROY_CUSTOMARY_JUSTICE } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { has_completed_focus = MAD_POL_ROY_ONE_MADAGASCAR_MANY_HOUSES } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { has_completed_focus = MAD_POL_AREMA_FOKONOLONA } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { has_completed_focus = MAD_POL_AREMA_FOOD_FIRST } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { has_completed_focus = MAD_POL_AREMA_PROVINCES } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { has_completed_focus = MAD_POL_MIL_SOUTHERN_SECURITY } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { has_completed_focus = MAD_POL_MIL_GENDARMERIE_REFORM } add_to_temp_variable = { MAD_rc_monthly_gain = 0.50 } }
	if = { limit = { has_completed_focus = MAD_POL_MIL_ENGINEERS } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { has_completed_focus = MAD_POL_TGV_INFRASTRUCTURE_FIRST } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { has_completed_focus = MAD_POL_TGV_HEALTH_FOR_ALL } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { has_completed_focus = MAD_POL_TGV_JIRAMA_DELIVERY } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { has_completed_focus = MAD_POL_TGV_FOOD_SOVEREIGNTY } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	# Successful resolution of the southern drought leaves a durable recovery dividend.
	if = { limit = { has_completed_focus = MAD_Thriving_south } add_to_temp_variable = { MAD_rc_monthly_gain = 0.50 } }

	# Completed rural access projects give their own permanent dividend; a project under construction does not.
	if = { limit = { has_country_flag = MAD_JIRAMA_RURAL_GRID_BUILT } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { has_country_flag = MAD_LEAD_ACCESS_BUILT } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }
	if = { limit = { has_country_flag = MAD_JIRAMA_DECIM_SOLAR_BUILT } add_to_temp_variable = { MAD_rc_monthly_gain = 0.25 } }

	# Cap positive recovery, even if several route and infrastructure milestones stack.
	if = {
		limit = { check_variable = { MAD_rc_monthly_gain > 2.99 } }
		set_variable = { MAD_rural_cooperation_cap_adjustment = MAD_rc_monthly_gain }
		subtract_from_variable = { MAD_rural_cooperation_cap_adjustment = 3.0 }
		set_temp_variable = { MAD_rc_monthly_gain = 3.0 }
	}
	set_variable = { MAD_rural_cooperation_monthly_gain = MAD_rc_monthly_gain }
	set_variable = { MAD_rural_cooperation_monthly_change = MAD_rc_monthly_gain }
	# The retained Daewoo lease drains Cooperation after the normal recovery calculation.
	if = {
		limit = { has_country_flag = MAD_DAEWOO_OUTCOME_WELL }
		add_to_variable = { MAD_rural_cooperation_monthly_change = -0.25 }
	}
	if = {
		limit = { has_idea = MAD_MADARAIL_COMAZAR_FREIGHT_CONCESSION }
		add_to_variable = { MAD_rural_cooperation_monthly_change = -0.25 }
	}

	# Net monthly change includes Daewoo/Comazar deductions and may fall to -1.00,
	# but never lower. Positive gross recovery remains capped at +3.00 above.
	if = {
		limit = { check_variable = { MAD_rural_cooperation_monthly_change < -1.00 } }
		set_variable = { MAD_rural_cooperation_monthly_change = -1.00 }
	}
	# Report the expected realised change after the 0-100 meter limits.
	set_temp_variable = { MAD_rc_cooperation_headroom = 100 }
	subtract_from_temp_variable = { MAD_rc_cooperation_headroom = MAD_rural_cooperation }
	if = {
		limit = { check_variable = { MAD_rural_cooperation_monthly_change > MAD_rc_cooperation_headroom } }
		set_variable = { MAD_rural_cooperation_meter_upper_adjustment = MAD_rural_cooperation_monthly_change }
		subtract_from_variable = { MAD_rural_cooperation_meter_upper_adjustment = MAD_rc_cooperation_headroom }
		set_variable = { MAD_rural_cooperation_monthly_change = MAD_rc_cooperation_headroom }
	}
	set_temp_variable = { MAD_rc_cooperation_floor = 0 }
	subtract_from_temp_variable = { MAD_rc_cooperation_floor = MAD_rural_cooperation }
	if = {
		limit = { check_variable = { MAD_rural_cooperation_monthly_change < MAD_rc_cooperation_floor } }
		set_variable = { MAD_rural_cooperation_meter_lower_adjustment = MAD_rc_cooperation_floor }
		subtract_from_variable = { MAD_rural_cooperation_meter_lower_adjustment = MAD_rural_cooperation_monthly_change }
		set_variable = { MAD_rural_cooperation_monthly_change = MAD_rc_cooperation_floor }
	}
	# The monthly tick applies this net amount, so keep gain and displayed change aligned.
	set_variable = { MAD_rural_cooperation_monthly_gain = MAD_rural_cooperation_monthly_change }
}

MAD_rural_cooperation_monthly_tick = {
	MAD_rural_cooperation_initialize = yes
	MAD_rural_cooperation_calculate_monthly_change = yes
	# Apply the already-calculated net monthly change (including lease deductions).
	add_to_variable = { MAD_rural_cooperation = MAD_rural_cooperation_monthly_gain }
	clamp_variable = { var = MAD_rural_cooperation min = 0 max = 100 }
	# After the meter changes, refresh the displayed rate for the next monthly pulse.
	MAD_rural_cooperation_calculate_monthly_change = yes
}

# Used only by explicitly listed targeted relief decisions. Scale their helpful negative
# deltas before the Dahalo interception effect is applied; cash costs and containment counts
# are deliberately unaffected.
MAD_rural_cooperation_boost_state_relief = {
	# Cooperation changes the effectiveness of targeted drought/famine relief:
	# below 35: -50%; 35-59.99: -20%; 60-79.99: no modifier; 80+: +20%.
	set_temp_variable = { MAD_rc_relief_multiplier = 0.50 }
	if = {
		limit = { ROOT = { check_variable = { MAD_rural_cooperation > 79.99 } } }
		set_temp_variable = { MAD_rc_relief_multiplier = 1.20 }
	}
	else_if = {
		limit = { ROOT = { check_variable = { MAD_rural_cooperation > 59.99 } } }
		set_temp_variable = { MAD_rc_relief_multiplier = 1.00 }
	}
	else_if = {
		limit = { ROOT = { check_variable = { MAD_rural_cooperation > 34.99 } } }
		set_temp_variable = { MAD_rc_relief_multiplier = 0.80 }
	}
	if = {
		limit = { check_variable = { MAD_dr_state_change < -0.001 } }
		multiply_temp_variable = { MAD_dr_state_change = MAD_rc_relief_multiplier }
	}
	if = {
		limit = { check_variable = { MAD_dr_state_famine < -0.001 } }
		multiply_temp_variable = { MAD_dr_state_famine = MAD_rc_relief_multiplier }
	}
}
