# remains party leader after their general character has been retired.

	clr_country_flag = MAD_MIL_PARTY_LEADER_GENERAL_STAFF_ANDRIANOME

		limit = {
			date > 2022.1.1
			has_character = MAD_michael_randrianirina
		}
		set_country_flag = MAD_MIL_PARTY_LEADER_CAPSAT_MICHAEL
	}
	else_if = {
		limit = { has_character = MAD_noel_rakotonandrasana }
		set_country_flag = MAD_MIL_PARTY_LEADER_CAPSAT_NOEL
	}
	else_if = {
		limit = {
			date > 2009.9.3
			has_character = MAD_lylison_rene
		}
		set_country_flag = MAD_MIL_PARTY_LEADER_CAPSAT_LYLISON
	}
	else_if = {
		limit = {
			date > 2009.9.3
			has_character = MAD_charles_andrianasoavina
		}
		set_country_flag = MAD_MIL_PARTY_LEADER_CAPSAT_CHARLES
	}
	else_if = {
		limit = {
			date > 2009.3.10
			has_character = MAD_andre_andriarijaona
		}
		set_country_flag = MAD_MIL_PARTY_LEADER_CAPSAT_ANDRE
	}
	else = {
		set_country_flag = MAD_MIL_PARTY_LEADER_CAPSAT_NOEL
	}
	set_country_flag = MAD_MIL_PARTY_LEADER_CAPSAT_SELECTED
}

set_leader_MAD = {
	# Madagascar-specific party names are defined in MD_politics_view_parties_l_english.yml.
	# These are the only 10 labelled MAD party indices; do not fill unrelated generic MD slots.

	# 1 — HVM (Conservatism). HVM only receives its named leader after the formation event.
	if = {
		limit = { check_variable = { ruling_party = 1 } }
		if = {
			limit = {
				has_country_flag = MAD_HVM_FORMED
				check_variable = { conservatism_leader = 0 }
			}
			add_to_variable = { conservatism_leader = 1 }
			hidden_effect = { kill_country_leader = yes }
			create_country_leader = {
				name = "Hery Rajaonarimampianina"
				picture = "Hery_Rajaonarimampianina.dds"
				ideology = conservatism
				traits = {
					western_conservatism
					career_politician
				}
			}
			if = { limit = { has_country_flag = do_not_retire } subtract_from_variable = { conservatism_leader = 1 } }
			set_temp_variable = { b = 1 }
		}
	}
	# 2 — TIM (Liberalism).
	else_if = {
		limit = { check_variable = { ruling_party = 2 } }
		if = { limit = { check_variable = { liberalism_leader = 0 } }
			add_to_variable = { liberalism_leader = 1 }
			hidden_effect = { kill_country_leader = yes }
			create_country_leader = {
				name = "Marc Ravalomanana"

						emerging_anarchist_communism
						career_politician
					}
				}
			}
			else_if = {
				limit = { has_country_flag = MAD_AREMA_RECONCILIATION }
				create_country_leader = {
					name = "Pierre Raharijaona"
					picture = "Pierre_Raharijaona.dds"
					ideology = anarchist_communism
					traits = {
						emerging_anarchist_communism
						career_politician
					}
				}
			}
