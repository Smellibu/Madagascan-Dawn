	#  - the suitor's concession decisions below raise MAD_madarail_confidence
	#  - with confidence >= 75 the player can complete the mission by hand -> STAY (concessions granted so far are kept)
	#  - if the 150 days run out, event mad_madarail.4 sends the suitor away; the player picks the next bidder
	#  - after the third suitor's 150 days, mad_madarail.4 forces the investors out -> DESIST

		icon = generic_political_discourse

		selectable_mission = yes

		activation = { always = no }

			custom_trigger_tooltip = {
				tooltip = MAD_MADARAIL_CONFIDENCE_MET_tt

	# Five recurring programmes and ten single-use policy decisions.
	# Recurring programme bonuses are applied by timed ideas only (180 days).
	# ============================================================================

	# ------------------------------------------------------------------------
	# REPEATABLE PROGRAMMES
	# Three cycles maximum per programme. Each cycle consumes RC and treasury.
	# The completion event lets the player choose a public or private delivery
	# model. Only the ownership BOP adjustment persists; the programme bonus is
	# a timed idea lasting 180 days.
	# ------------------------------------------------------------------------

		icon = generic_construction
		visible = { always = yes }
		available = {
						custom_trigger_tooltip = {
				tooltip = MAD_POLICY_RC_REQUIRE_4_TT
				check_variable = { MAD_rural_cooperation > 3.99 }
			}
		}
		cost = 35
		fire_only_once = no
		# Native decision timer: the programme runs for 365 days.
		days_remove = 365
		custom_cost_trigger = { always = yes }
		custom_cost_text = cost_0_75
		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Regional road maintenance programme launched"
			set_temp_variable = { treasury_change = -0.75 }

				limit = { NOT = { state = 267 } }
				add_extra_state_shared_building_slots = 1
			}
		}
		ai_will_do = { base = 0 }
	}
