# .30 the rains fail in one region   .31 a cyclone (rain, but the wrong kind)   .32 a state tips into famine
# .33 the wells run dry in one state
#
# These are what make the map move unevenly: a countrywide number would never show you that Androy is finished
# while Analamanga is fine.

# The rains fail somewhere. Picks the driest state that is not already at the top of the scale.

	id = mad_drought.30
	title = mad_drought.30.t
	desc = mad_drought.30.d
	picture = GFX_report_event_economy_default
	is_triggered_only = yes

	option = {
		name = mad_drought.30.a
		log = "[GetDateText]: [Root.GetName]: mad_drought.30 - the rains fail in one region"
		custom_effect_tooltip = mad_drought.30_tt
		hidden_effect = {
			random_owned_state = {
				limit = { check_variable = { MAD_state_drought < 85 } }
				set_temp_variable = { MAD_dr_state_change = 9 }
				MAD_drought_state_apply_change = yes
				set_temp_variable = { MAD_dr_state_famine = 5 }
				MAD_drought_state_apply_famine = yes
			}
			MAD_drought_recalc_national = yes
			add_to_variable = { MAD_drought_shock_pressure = 0.25 }
		}
	}
}
