########################################################################
# VANILLA EXPORT MARKET CYCLE
#
# Started by MAD_VANILLA_BOOM_BUST_CYCLE.
# One event handles both the boom and bust states. The player gets a small
# response decision at each shock, so the branch plays as:
#
#   export policy -> market shock -> player response -> next shock
#
# MAD_vanilla_shock_state:
#    0   = no previous temporary shock
#   11   = boom / reinvest
#   12   = boom / bank profits
#  -11   = bust / support growers
#  -12   = bust / let market correct
########################################################################

add_namespace = mad_vanilla

country_event = {
	id = mad_vanilla.1
	title = mad_vanilla.1.t
	desc = {
		text = mad_vanilla.1.d_boom
		trigger = { has_country_flag = MAD_VANILLA_PRICE_BOOM }
	}
	desc = {
		text = mad_vanilla.1.d_bust
		trigger = { has_country_flag = MAD_VANILLA_PRICE_BUST }
	}
	picture = GFX_report_event_economy_default
	is_triggered_only = yes
	trigger = { has_country_flag = MAD_VANILLA_VOLATILE_MARKET }

	immediate = {
		log = "[GetDateText]: [Root.GetName]: mad_vanilla.1 - vanilla market shock"

		# Remove the previous temporary market shock before applying the next one.
		# Stability and political-power changes from the previous response are
		# deliberately one-off and are not reversed.
		if = {
			limit = { check_variable = { MAD_vanilla_shock_state = 11 } }
			add_to_variable = { MAD_economy_international_market_income_modifier = -0.10 tooltip = international_market_income_modifier_tt }
			add_to_variable = { MAD_economy_return_on_investment_modifier = -0.04 tooltip = return_on_investment_modifier_tt }
		}
		else_if = {
			limit = { check_variable = { MAD_vanilla_shock_state = 12 } }
			add_to_variable = { MAD_economy_international_market_income_modifier = -0.06 tooltip = international_market_income_modifier_tt }
			add_to_variable = { MAD_economy_return_on_investment_modifier = -0.02 tooltip = return_on_investment_modifier_tt }
			add_to_variable = { MAD_economy_expected_welfare_modifier = -0.25 tooltip = expected_welfare_modifier_tt }
		}
		else_if = {
			limit = { check_variable = { MAD_vanilla_shock_state = -11 } }
			add_to_variable = { MAD_economy_international_market_income_modifier = 0.04 tooltip = international_market_income_modifier_tt }
