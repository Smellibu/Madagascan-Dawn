country_event = {
	id = mad_diplomacy.13
	hidden = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = {
				OR = {
					AND = {
						has_country_flag = MAD_dip_mrt_ioc_accepted
						has_country_flag = MAD_dip_com_ioc_accepted
					}
					AND = {
						has_country_flag = MAD_dip_mrt_ioc_accepted
						has_country_flag = MAD_dip_sey_ioc_accepted
					}
					AND = {
						has_country_flag = MAD_dip_com_ioc_accepted
						has_country_flag = MAD_dip_sey_ioc_accepted
					}
				}
			}
			if = {
				limit = { has_idea = MAD_dip_ioc_network }
				swap_ideas = {
					remove_idea = MAD_dip_ioc_network
					add_idea = MAD_dip_ioc_active
				}
			}
			else_if = {
				limit = { NOT = { has_idea = MAD_dip_ioc_active } }
				add_ideas = MAD_dip_ioc_active
			}
		}
		else = {
			if = {
				limit = {
					NOT = { has_idea = MAD_dip_ioc_network }
					NOT = { has_idea = MAD_dip_ioc_active }
				}
				add_ideas = MAD_dip_ioc_network
			}
		}
	}
}
